@Entity
@Table(name = "restaurant_orders")
public class RestaurantOrder {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(name = "order_number", nullable = false, unique = true, length = 50)
    private String orderNumber;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "table_id", nullable = false)
    private RestaurantTable restaurantTable;

    @Enumerated(EnumType.STRING)
    @Column(nullable = false, length = 30)
    private OrderStatus status = OrderStatus.PENDING;

    @Column(name = "total_amount", nullable = false, precision = 10, scale = 2)
    private BigDecimal totalAmount = BigDecimal.ZERO;

    @Column(name = "total_preparation_work_minutes", nullable = false)

    @Column(name = "customer_note", length = 500)

    private boolean inventoryDeducted = false;

    @Column(name = "created_at", nullable = false, updatable = false)
    private Instant createdAt;

    @Column(name = "updated_at", nullable = false)
    private Instant updatedAt;

    @OneToMany(
        mappedBy = "order",

package com.khaidai.backend.service;

import com.khaidai.backend.dto.InventoryOverviewResponse;
import com.khaidai.backend.dto.InventoryRequests;

import com.khaidai.backend.entity.InventoryIngredient;
import com.khaidai.backend.entity.StockMovement;
import com.khaidai.backend.entity.StockMovementType;

import com.khaidai.backend.repository.InventoryIngredientRepository;

import org.springframework.beans.factory.annotation.Value;

@Service
public class InventoryService {

    private static final Set<String> ALLOWED_UNITS = Set.of(
            "g", "kg", "ml", "L", "pcs", "pack", "bottle", "box"
    );

    private final InventoryIngredientRepository ingredientRepository;
    private final StockMovementRepository movementRepository;

    @Value("${khaidai.admin.api-key:}")
    private String configuredAdminKey;

    public InventoryService(
            InventoryIngredientRepository ingredientRepository,
            StockMovementRepository movementRepository
    ) {
        this.ingredientRepository = ingredientRepository;
        this.movementRepository = movementRepository;
    }

    // ==========================================
    // GET INVENTORY OVERVIEW
    // ==========================================

    @Transactional(readOnly = true)
    public InventoryOverviewResponse getOverview(String adminKey) {

        validateAdminKey(adminKey);

        List<InventoryOverviewResponse.IngredientRow> ingredients =
                ingredientRepository.findAllByOrderByNameAsc()
                        .stream()
                        .map(this::toIngredientRow)
                        .toList();

        List<InventoryOverviewResponse.MovementRow> movements =
                movementRepository.findTop50ByOrderByCreatedAtDesc()
                        .stream()
                        .map(this::toMovementRow)
                        .toList();

        return new InventoryOverviewResponse(
                ingredients,
                movements
        );
    }

    // ==========================================
    // CREATE INGREDIENT
    // ==========================================

    @Transactional
    public void createIngredient(
            InventoryRequests.CreateIngredientRequest request,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        if (request == null) {
            throw badRequest("Ingredient information is required.");
        }

        String name = cleanName(request.name());
        String unit = validateUnit(request.unit());

        BigDecimal openingStock = nonNegative(
                request.openingStock(),
                "Opening stock"
        );

        BigDecimal reorderLevel = nonNegative(
                request.reorderLevel(),
                "Reorder level"
        );

        if (ingredientRepository.findByNameIgnoreCase(name).isPresent()) {
            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "An ingredient with this name already exists."
            );
        }

        InventoryIngredient ingredient = new InventoryIngredient();

        ingredient.setName(name);
        ingredient.setUnit(unit);
        ingredient.setCurrentStock(openingStock);
        ingredient.setReorderLevel(reorderLevel);
        ingredient.setActive(true);

        ingredientRepository.saveAndFlush(ingredient);

        if (openingStock.compareTo(BigDecimal.ZERO) > 0) {

            movementRepository.save(
                    new StockMovement(
                            ingredient,
                            StockMovementType.OPENING,
                            openingStock,
                            openingStock,
                            "Opening stock"
                    )
            );
        }
    }

    // ==========================================
    // UPDATE INGREDIENT
    // ==========================================

    @Transactional
    public void updateIngredient(
            Long id,
            InventoryRequests.UpdateIngredientRequest request,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        if (request == null) {
            throw badRequest("Update information is required.");
        }

        InventoryIngredient ingredient = findForUpdate(id);

        if (request.name() != null) {

            String name = cleanName(request.name());

            ingredientRepository.findByNameIgnoreCase(name)
                    .filter(existing -> !existing.getId().equals(id))
                    .ifPresent(existing -> {
                        throw new ResponseStatusException(
                                HttpStatus.CONFLICT,
                                "Another ingredient already uses this name."
                        );
                    });

            ingredient.setName(name);
        }

        if (request.unit() != null) {
            ingredient.setUnit(validateUnit(request.unit()));
        }

        if (request.reorderLevel() != null) {
            ingredient.setReorderLevel(
                    nonNegative(
                            request.reorderLevel(),
                            "Reorder level"
                    )
            );
        }

        if (request.active() != null) {
            ingredient.setActive(request.active());
        }

        ingredientRepository.save(ingredient);
    }

    // ==========================================
    // RECORD STOCK MOVEMENT
    // ==========================================

    @Transactional
    public void recordMovement(
            Long id,
            InventoryRequests.StockMovementRequest request,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        if (request == null || request.type() == null) {
            throw badRequest("Stock movement information is required.");
        }

        InventoryIngredient ingredient = findForUpdate(id);

        if (!ingredient.isActive()) {
            throw badRequest("This ingredient is inactive.");
        }

        StockMovementType type;

        try {
            type = StockMovementType.valueOf(
                    request.type().trim().toUpperCase(Locale.ROOT)
            );
        } catch (IllegalArgumentException exception) {
            throw badRequest("Invalid stock movement type.");
        }

        if (type == StockMovementType.OPENING) {
            throw badRequest("Opening stock is only allowed when creating an ingredient.");
        }

        BigDecimal quantity = nonNegative(
                request.quantity(),
                "Quantity"
        );

        if (quantity.compareTo(BigDecimal.ZERO) == 0 &&
                type != StockMovementType.ADJUSTMENT) {
            throw badRequest("Quantity must be greater than zero.");
        }

        BigDecimal current = ingredient.getCurrentStock();

        BigDecimal change;
        BigDecimal newBalance;

        switch (type) {

            case STOCK_IN -> {

                change = quantity;
                newBalance = current.add(quantity);
            }

            case WASTE -> {

                change = quantity.negate();
                newBalance = current.subtract(quantity);

                if (newBalance.compareTo(BigDecimal.ZERO) < 0) {
                    throw new ResponseStatusException(
                            HttpStatus.CONFLICT,
                            "Waste quantity exceeds available stock."
                    );
                }
            }

            case ADJUSTMENT -> {

                // For adjustment, quantity means actual
                // physically counted stock, not a delta.

                newBalance = quantity;
                change = newBalance.subtract(current);
            }

            default -> throw badRequest("Unsupported stock movement.");
        }

        ingredient.setCurrentStock(newBalance);

        ingredientRepository.save(ingredient);

        String note = request.note() == null
                ? null
                : request.note().trim();

        if (note != null && note.isEmpty()) {
            note = null;
        }

        if (note != null && note.length() > 250) {
            throw badRequest("Movement note cannot exceed 250 characters.");
        }

        movementRepository.save(
                new StockMovement(
                        ingredient,
                        type,
                        change,
                        newBalance,
                        note
                )
        );
    }

    // ==========================================
    // MAPPING
    // ==========================================

    private InventoryOverviewResponse.IngredientRow toIngredientRow(
            InventoryIngredient ingredient
    ) {

        boolean lowStock =
                ingredient.getCurrentStock()
                        .compareTo(ingredient.getReorderLevel()) <= 0;

        return new InventoryOverviewResponse.IngredientRow(

                ingredient.getId(),
                ingredient.getName(),
                ingredient.getUnit(),

                ingredient.getCurrentStock(),
                ingredient.getReorderLevel(),

                ingredient.isActive(),
                lowStock,

                ingredient.getUpdatedAt()
        );
    }

    private InventoryOverviewResponse.MovementRow toMovementRow(
            StockMovement movement
    ) {

        InventoryIngredient ingredient = movement.getIngredient();

        return new InventoryOverviewResponse.MovementRow(

                movement.getId(),
                ingredient.getId(),

                ingredient.getName(),
                ingredient.getUnit(),

                movement.getMovementType().name(),

                movement.getQuantityChange(),
                movement.getBalanceAfter(),

                movement.getNote(),
                movement.getCreatedAt()
        );
    }

    // ==========================================
    // VALIDATION
    // ==========================================

    private InventoryIngredient findForUpdate(Long id) {

        if (id == null) {
            throw badRequest("Ingredient ID is required.");
        }

        return ingredientRepository.findByIdForUpdate(id)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Ingredient not found."
                ));
    }

    private String cleanName(String value) {

        if (value == null || value.isBlank()) {
            throw badRequest("Ingredient name is required.");
        }

        String name = value.trim();

        if (name.length() > 100) {
            throw badRequest("Ingredient name cannot exceed 100 characters.");
        }

        return name;
    }

    private String validateUnit(String value) {

        if (value == null || !ALLOWED_UNITS.contains(value.trim())) {
            throw badRequest(
                    "Unit must be g, kg, ml, L, pcs, pack, bottle or box."
            );
        }

        return value.trim();
    }

    private BigDecimal nonNegative(
            BigDecimal value,
            String field
    ) {

        if (value == null || value.compareTo(BigDecimal.ZERO) < 0) {
            throw badRequest(field + " cannot be negative.");
        }

        return value;
    }

    private void validateAdminKey(String providedKey) {

        if (configuredAdminKey == null ||
                configuredAdminKey.isBlank()) {

            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "Admin API key is not configured."
            );
        }

        if (providedKey == null ||
                !MessageDigest.isEqual(
                        configuredAdminKey.getBytes(StandardCharsets.UTF_8),
                        providedKey.getBytes(StandardCharsets.UTF_8)
                )) {

            throw new ResponseStatusException(
                    HttpStatus.UNAUTHORIZED,
                    "Invalid administrator key."
            );
        }
    }

    private ResponseStatusException badRequest(String message) {

        return new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                message
        );
    }
}

package com.khaidai.backend.service;

import com.khaidai.backend.dto.AdminOrderResponse;
import com.khaidai.backend.entity.OrderStatus;
import com.khaidai.backend.entity.Payment;

import com.khaidai.backend.entity.RestaurantOrder;
import com.khaidai.backend.repository.OrderRepository;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.util.List;
import java.util.Locale;
import java.util.UUID;

@Service
public class OrderAdminService {

    private final OrderRepository orderRepository;
    private final PaymentRepository paymentRepository;
    private final OrderInventoryService orderInventoryService;

    @Value("${khaidai.admin.api-key:}")
    private String configuredAdminKey;

    public OrderAdminService(
            OrderRepository orderRepository,
            PaymentRepository paymentRepository,
            OrderInventoryService orderInventoryService
    ) {
        this.orderRepository = orderRepository;
        this.paymentRepository = paymentRepository;
        this.orderInventoryService = orderInventoryService;
    }

    @Transactional(readOnly = true)
    public List<AdminOrderResponse> getAllOrders(String adminKey) {

        validateAdminKey(adminKey);

        return orderRepository
                .findAllByOrderByCreatedAtDesc()
                .stream()
                .map(this::toResponse)
                .toList();
    }

    @Transactional
    public AdminOrderResponse updateOrderStatus(
            String orderNumber,
            String requestedStatus,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        if (requestedStatus == null || requestedStatus.isBlank()) {
            throw badRequest("Order status is required.");
        }

        OrderStatus newStatus;

        try {
            newStatus = OrderStatus.valueOf(
                    requestedStatus.trim().toUpperCase(Locale.ROOT)
            );
        } catch (IllegalArgumentException exception) {
            throw badRequest("Invalid order status.");
        }

        RestaurantOrder order = findOrder(orderNumber);

        if (!isValidTransition(order.getStatus(), newStatus)) {
            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "Cannot change order status from "
                            + order.getStatus() + " to " + newStatus + "."
            );
        }

        if (newStatus == OrderStatus.ACCEPTED) {
            orderInventoryService.validateAvailability(order);
        } else if (newStatus == OrderStatus.PREPARING) {
            orderInventoryService.deductForOrder(order);
        } else if (newStatus == OrderStatus.CANCELLED) {
            orderInventoryService.restoreForOrder(order);
        }

        order.setStatus(newStatus);

        RestaurantOrder savedOrder = orderRepository.save(order);

        return toResponse(savedOrder);
    }

    // ==========================================
    // MARK CASH PAYMENT AS PAID
    // ==========================================

    @Transactional
    public AdminOrderResponse markCashPaymentAsPaid(
            String orderNumber,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        RestaurantOrder order = findOrder(orderNumber);

        if (order.getStatus() == OrderStatus.AWAITING_PAYMENT) {
            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "This order is awaiting online payment."
            );
        }

        List<Payment> existingPayments =
                paymentRepository.findAllByOrder_IdOrderByCreatedAtDesc(
                        order.getId()
                );

        Payment latest = existingPayments.isEmpty()
                ? null
                : existingPayments.get(0);

        if (latest != null &&
                latest.getStatus() == PaymentStatus.PAID) {

            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "This order is already marked as paid."
            );
        }

        if (latest != null &&
                latest.getPaymentMethod() == PaymentMethod.SSLCOMMERZ &&
                latest.getStatus() == PaymentStatus.PENDING) {

            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "An online payment is still pending."
            );
        }

        Payment cashPayment = new Payment();

        cashPayment.setOrder(order);

        cashPayment.setTransactionId(
                "CASH-" + UUID.randomUUID()
                        .toString()
                        .replace("-", "")
                        .substring(0, 24)
                        .toUpperCase(Locale.ROOT)
        );

        cashPayment.setPaymentMethod(PaymentMethod.CASH);
        cashPayment.setStatus(PaymentStatus.PAID);
        cashPayment.setAmount(order.getTotalAmount());
        cashPayment.setCurrency("BDT");

        paymentRepository.saveAndFlush(cashPayment);

        return toResponse(order);
    }

    private RestaurantOrder findOrder(String orderNumber) {

        return orderRepository
                .findByOrderNumberForUpdate(orderNumber)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Order not found."
                ));
    }

    private boolean isValidTransition(
            OrderStatus current,
            OrderStatus next
    ) {

        return switch (current) {

            case PENDING ->
                    next == OrderStatus.ACCEPTED ||
                    next == OrderStatus.REJECTED ||
                    next == OrderStatus.CANCELLED;

            case ACCEPTED ->
                    next == OrderStatus.PREPARING ||
                    next == OrderStatus.CANCELLED;

            case PREPARING ->
                    next == OrderStatus.READY ||
                    next == OrderStatus.CANCELLED;

            case READY ->
                    next == OrderStatus.SERVED;

            default -> false;
        };
    }

    private void validateAdminKey(String providedAdminKey) {

        if (configuredAdminKey == null ||
                configuredAdminKey.isBlank()) {

            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "Admin API key is not configured."
            );
        }

        if (providedAdminKey == null ||
                !MessageDigest.isEqual(
                        configuredAdminKey.getBytes(StandardCharsets.UTF_8),
                        providedAdminKey.getBytes(StandardCharsets.UTF_8)
                )) {

            throw new ResponseStatusException(
                    HttpStatus.UNAUTHORIZED,
                    "Invalid admin API key."
            );
        }
    }

    // ==========================================
    // MAP ORDER AND LATEST PAYMENT
    // ==========================================

    private AdminOrderResponse toResponse(RestaurantOrder order) {

        List<Payment> payments =
                paymentRepository.findAllByOrder_IdOrderByCreatedAtDesc(
                        order.getId()
                );

        Payment latest = payments.isEmpty()
                ? null
                : payments.get(0);

        String paymentMethod = latest != null
                ? latest.getPaymentMethod().name()
                : order.getStatus() == OrderStatus.AWAITING_PAYMENT
                    ? "SSLCOMMERZ"
                    : "CASH";

        String paymentStatus = latest != null
                ? latest.getStatus().name()
                : "CASH".equals(paymentMethod)
                    ? "CASH_DUE"
                    : "PENDING";

        List<AdminOrderResponse.ItemResponse> items =
                order.getItems()
                        .stream()
                        .map(item ->
                                new AdminOrderResponse.ItemResponse(
                                        item.getFoodId(),
                                        item.getFoodName(),
                                        item.getQuantity(),
                                        item.getUnitPrice(),
                                        item.getLineTotal(),
                                        item.getSpiceLevel(),
                                        item.getPreparationTimeMinutes(),
                                        item.getSelectedAddons()
                                                .stream()
                                                .map(addon ->
                                                        new AdminOrderResponse.AddonResponse(
                                                                addon.getName(),
                                                                addon.getPrice()
                                                        )
                                                )
                                                .toList()
                                )
                        )
                        .toList();

        return new AdminOrderResponse(
                order.getId(),
                order.getOrderNumber(),
                order.getRestaurantTable().getTableNumber(),
                order.getStatus(),
                order.getTotalAmount(),
                order.getTotalPreparationWorkMinutes(),
                order.getCustomerNote(),
                order.getCreatedAt(),
                order.getUpdatedAt(),

                paymentMethod,
                paymentStatus,
                latest != null ? latest.getTransactionId() : null,
                latest != null ? latest.getAmount() : null,
                latest != null ? latest.getCurrency() : "BDT",
                latest != null ? latest.getPaidAt() : null,

                items
        );
    }

    private ResponseStatusException badRequest(String message) {

        return new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                message
        );
    }
}

package com.khaidai.backend.service;

import com.khaidai.backend.entity.InventoryIngredient;
import com.khaidai.backend.entity.MenuRecipeIngredient;
import com.khaidai.backend.entity.OrderItem;
import com.khaidai.backend.entity.RestaurantOrder;
import com.khaidai.backend.entity.StockMovement;
import com.khaidai.backend.entity.StockMovementType;

import com.khaidai.backend.repository.InventoryIngredientRepository;
import com.khaidai.backend.repository.MenuRecipeIngredientRepository;
import com.khaidai.backend.repository.StockMovementRepository;

import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;

import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Set;
import java.util.TreeMap;
import java.util.TreeSet;

@Service
public class OrderInventoryService {

    private final InventoryIngredientRepository ingredientRepository;
    private final MenuRecipeIngredientRepository recipeRepository;
    private final StockMovementRepository movementRepository;

    public OrderInventoryService(
            InventoryIngredientRepository ingredientRepository,
            MenuRecipeIngredientRepository recipeRepository,
            StockMovementRepository movementRepository
    ) {
        this.ingredientRepository = ingredientRepository;
        this.recipeRepository = recipeRepository;
        this.movementRepository = movementRepository;
    }

    // ==========================================
    // CHECK STOCK BEFORE ACCEPTING AN ORDER
    // ==========================================

    @Transactional
    public void validateAvailability(RestaurantOrder order) {

        Map<Long, BigDecimal> required =
                calculateRequiredIngredients(order);

        Map<Long, InventoryIngredient> locked =
                lockIngredients(required.keySet());

        validateStock(required, locked);
    }

    // ==========================================
    // DEDUCT STOCK WHEN PREPARATION STARTS
    // ==========================================

    @Transactional
    public void deductForOrder(RestaurantOrder order) {

        if (order.isInventoryDeducted()) {
            return;
        }

        Map<Long, BigDecimal> required =
                calculateRequiredIngredients(order);

        Map<Long, InventoryIngredient> locked =
                lockIngredients(required.keySet());

        validateStock(required, locked);

        for (Map.Entry<Long, BigDecimal> entry : required.entrySet()) {

            InventoryIngredient ingredient = locked.get(entry.getKey());

            BigDecimal quantity = entry.getValue();

            BigDecimal newBalance =
                    ingredient.getCurrentStock().subtract(quantity);

            ingredient.setCurrentStock(newBalance);

            ingredientRepository.save(ingredient);

            movementRepository.save(
                    new StockMovement(
                            ingredient,
                            StockMovementType.ORDER_DEDUCTION,
                            quantity.negate(),
                            newBalance,
                            "Deducted for order " + order.getOrderNumber(),
                            order.getOrderNumber()
                    )
            );
        }

        order.setInventoryDeducted(true);
    }

    // ==========================================
    // RESTORE STOCK WHEN A PREPARING ORDER
    // IS CANCELLED
    // ==========================================

    @Transactional
    public void restoreForOrder(RestaurantOrder order) {

        if (!order.isInventoryDeducted()) {
            return;
        }

        List<StockMovement> deductions =
                movementRepository
                        .findAllByReferenceAndMovementTypeOrderByIdAsc(
                                order.getOrderNumber(),
                                StockMovementType.ORDER_DEDUCTION
                        );

        if (deductions.isEmpty()) {
            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "Inventory deduction history is missing. Stock cannot be safely restored."
            );
        }

        Map<Long, BigDecimal> restorationAmounts = new TreeMap<>();

        for (StockMovement deduction : deductions) {

            BigDecimal change = deduction.getQuantityChange();

            if (change == null ||
                    change.compareTo(BigDecimal.ZERO) >= 0) {

                throw new ResponseStatusException(
                        HttpStatus.CONFLICT,
                        "Invalid inventory deduction history."
                );
            }

            Long ingredientId = deduction.getIngredient().getId();

            restorationAmounts.merge(
                    ingredientId,
                    change.negate(),
                    BigDecimal::add
            );
        }

        Map<Long, InventoryIngredient> locked =
                lockIngredients(restorationAmounts.keySet());

        for (Map.Entry<Long, BigDecimal> entry :
                restorationAmounts.entrySet()) {

            InventoryIngredient ingredient = locked.get(entry.getKey());

            BigDecimal quantity = entry.getValue();

            BigDecimal newBalance =
                    ingredient.getCurrentStock().add(quantity);

            ingredient.setCurrentStock(newBalance);

            ingredientRepository.save(ingredient);

            movementRepository.save(
                    new StockMovement(
                            ingredient,
                            StockMovementType.ORDER_RESTORATION,
                            quantity,
                            newBalance,
                            "Restored after cancellation of order "
                                    + order.getOrderNumber(),
                            order.getOrderNumber()
                    )
            );
        }

        order.setInventoryDeducted(false);
    }

    // ==========================================
    // CALCULATE TOTAL REQUIRED INGREDIENTS
    // ==========================================

    private Map<Long, BigDecimal> calculateRequiredIngredients(
            RestaurantOrder order
    ) {

        if (order == null ||
                order.getItems() == null ||
                order.getItems().isEmpty()) {

            throw conflict("Order has no items.");
        }

        List<Integer> foodIds = order.getItems()
                .stream()
                .map(OrderItem::getFoodId)
                .filter(Objects::nonNull)
                .distinct()
                .toList();

        List<MenuRecipeIngredient> recipeRows =
                foodIds.isEmpty()
                        ? List.of()
                        : recipeRepository.findAllForFoodIds(foodIds);

        Map<Integer, List<MenuRecipeIngredient>> recipesByFood =
                new HashMap<>();

        for (MenuRecipeIngredient recipe : recipeRows) {

            recipesByFood.computeIfAbsent(
                    recipe.getFoodId(),
                    ignored -> new ArrayList<>()
            ).add(recipe);
        }

        Map<Long, BigDecimal> required = new TreeMap<>();

        for (OrderItem item : order.getItems()) {

            if (item.getFoodId() == null ||
                    item.getQuantity() == null ||
                    item.getQuantity() <= 0) {

                throw conflict("Invalid quantity or menu item in order.");
            }

            List<MenuRecipeIngredient> recipeLines =
                    recipesByFood.get(item.getFoodId());

            if (recipeLines == null || recipeLines.isEmpty()) {

                String foodName = item.getFoodName() == null
                        ? "Food ID " + item.getFoodId()
                        : item.getFoodName();

                throw conflict(
                        "No recipe configured for " + foodName
                                + ". Add its ingredients before accepting the order."
                );
            }

            for (MenuRecipeIngredient recipe : recipeLines) {

                InventoryIngredient ingredient = recipe.getIngredient();

                if (ingredient == null ||
                        recipe.getQuantityPerServing() == null ||
                        recipe.getQuantityPerServing()
                                .compareTo(BigDecimal.ZERO) <= 0) {

                    throw conflict("Invalid recipe configuration.");
                }

                BigDecimal needed =
                        recipe.getQuantityPerServing()
                                .multiply(BigDecimal.valueOf(item.getQuantity()));

                required.merge(
                        ingredient.getId(),
                        needed,
                        BigDecimal::add
                );
            }
        }

        return required;
    }

    // ==========================================
    // LOCK INGREDIENTS IN A CONSISTENT ORDER
    // ==========================================

    private Map<Long, InventoryIngredient> lockIngredients(
            Set<Long> ingredientIds
    ) {

        Map<Long, InventoryIngredient> locked =
                new LinkedHashMap<>();

        for (Long id : new TreeSet<>(ingredientIds)) {

            InventoryIngredient ingredient =
                    ingredientRepository.findByIdForUpdate(id)
                            .orElseThrow(() -> new ResponseStatusException(
                                    HttpStatus.NOT_FOUND,
                                    "Recipe ingredient no longer exists."
                            ));

            locked.put(id, ingredient);
        }

        return locked;
    }

    // ==========================================
    // VALIDATE STOCK
    // ==========================================

    private void validateStock(
            Map<Long, BigDecimal> required,
            Map<Long, InventoryIngredient> locked
    ) {

        for (Map.Entry<Long, BigDecimal> entry : required.entrySet()) {

            InventoryIngredient ingredient = locked.get(entry.getKey());

            if (!ingredient.isActive()) {

                throw conflict(
                        "Ingredient " + ingredient.getName()
                                + " is inactive. Update its recipe or activate it."
                );
            }

            BigDecimal available = ingredient.getCurrentStock();
            BigDecimal needed = entry.getValue();

            if (available.compareTo(needed) < 0) {

                throw conflict(
                        "Insufficient stock for " + ingredient.getName()
                                + ". Required: " + format(needed)
                                + " " + ingredient.getUnit()
                                + ", available: " + format(available)
                                + " " + ingredient.getUnit() + "."
                );
            }
        }
    }

    private String format(BigDecimal value) {
        return value.stripTrailingZeros().toPlainString();
    }

    private ResponseStatusException conflict(String message) {

        return new ResponseStatusException(
                HttpStatus.CONFLICT,
                message
        );
    }
}

import com.khaidai.backend.dto.RecipeOverviewResponse;

public class RecipeService {

    public RecipeService(

    // GET RECIPE OVERVIEW

    @Transactional(readOnly = true)
    public RecipeOverviewResponse getOverview(String adminKey) {

        validateAdminKey(adminKey);

        List<RecipeOverviewResponse.MenuItemOption> menuItems =
                MenuCatalog.all()
                        .stream()
                        .map(food -> new RecipeOverviewResponse.MenuItemOption(
                                food.id(),
                                food.name(),
                                food.price()
                        ))
                        .toList();

        List<RecipeOverviewResponse.IngredientOption> ingredients =
                ingredientRepository.findAllByOrderByNameAsc()
                        .stream()
                        .filter(InventoryIngredient::isActive)
                        .map(ingredient ->
                                new RecipeOverviewResponse.IngredientOption(
                                        ingredient.getId(),
                                        ingredient.getName(),
                                        ingredient.getUnit(),
                                        ingredient.getCurrentStock()
                                )
                        )
                        .toList();

        List<RecipeOverviewResponse.RecipeRow> recipes =
                recipeRepository.findAllWithIngredient()
                        .stream()
                        .map(this::toRecipeRow)
                        .toList();

        return new RecipeOverviewResponse(
                menuItems,
                ingredients,
                recipes
        );
    }

    // ==========================================
    // ADD INGREDIENT TO RECIPE
    // ==========================================

    @Transactional
    public void addIngredient(
            Integer foodId,
            RecipeRequests.AddIngredientRequest request,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        if (request == null || request.ingredientId() == null) {
            throw badRequest("Please select an ingredient.");
        }

        MenuCatalog.FoodItem food = MenuCatalog.findById(foodId);

        if (food == null) {
            throw badRequest("Invalid menu item.");
        }

        InventoryIngredient ingredient = ingredientRepository
                .findById(request.ingredientId())
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Ingredient not found."
                ));

        if (!ingredient.isActive()) {
            throw badRequest("This ingredient is inactive.");
        }

        if (recipeRepository.findByFoodIdAndIngredient_Id(
                foodId,
                ingredient.getId()
        ).isPresent()) {

            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "This ingredient is already in the recipe."
            );
        }

        BigDecimal quantity = positiveQuantity(
                request.quantityPerServing()
        );

        MenuRecipeIngredient recipe = new MenuRecipeIngredient();

        recipe.setFoodId(food.id());
        recipe.setFoodName(food.name());
        recipe.setIngredient(ingredient);
        recipe.setQuantityPerServing(quantity);

        recipeRepository.save(recipe);
    }

    // ==========================================
    // UPDATE RECIPE QUANTITY
    // ==========================================

    @Transactional
    public void updateIngredient(
            Long recipeId,
            RecipeRequests.UpdateIngredientRequest request,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        if (request == null) {
            throw badRequest("Recipe update information is required.");
        }

        MenuRecipeIngredient recipe = recipeRepository
                .findById(recipeId)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Recipe ingredient not found."
                ));

        recipe.setQuantityPerServing(
                positiveQuantity(request.quantityPerServing())
        );

        recipeRepository.save(recipe);
    }

    // ==========================================
    // REMOVE INGREDIENT FROM RECIPE
    // ==========================================

    @Transactional
    public void deleteIngredient(Long recipeId, String adminKey) {

        validateAdminKey(adminKey);

        MenuRecipeIngredient recipe = recipeRepository
                .findById(recipeId)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Recipe ingredient not found."
                ));

        recipeRepository.delete(recipe);
    }

    // ==========================================
    // RESPONSE MAPPING
    // ==========================================

    private RecipeOverviewResponse.RecipeRow toRecipeRow(
            MenuRecipeIngredient recipe
    ) {

        InventoryIngredient ingredient = recipe.getIngredient();

        return new RecipeOverviewResponse.RecipeRow(
                recipe.getId(),
                recipe.getFoodId(),
                recipe.getFoodName(),
                ingredient.getId(),
                ingredient.getName(),
                ingredient.getUnit(),
                ingredient.getCurrentStock(),
                recipe.getQuantityPerServing(),
                ingredient.isActive()
        );
    }

    // ==========================================
    // QUANTITY VALIDATION
    // ==========================================

    private BigDecimal positiveQuantity(BigDecimal value) {

        if (value == null || value.compareTo(BigDecimal.ZERO) <= 0) {
            throw badRequest("Quantity per serving must be greater than zero.");
        }

        BigDecimal normalized;

        try {
            normalized = value.setScale(3, RoundingMode.UNNECESSARY);
        } catch (ArithmeticException exception) {
            throw badRequest(
                    "Quantity supports a maximum of 3 decimal places."
            );
        }

        if (normalized.precision() > 14) {
            throw badRequest("Recipe quantity is too large.");
        }

        return normalized;
    }

    // ==========================================
    // ADMIN AUTHENTICATION
    // ==========================================

    private void validateAdminKey(String providedKey) {

        if (configuredAdminKey == null ||
                configuredAdminKey.isBlank()) {

            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "Admin API key is not configured."
            );
        }

        if (providedKey == null ||
                !MessageDigest.isEqual(
                        configuredAdminKey.getBytes(StandardCharsets.UTF_8),
                        providedKey.getBytes(StandardCharsets.UTF_8)
                )) {

            throw new ResponseStatusException(
                    HttpStatus.UNAUTHORIZED,
                    "Invalid administrator key."
            );
        }
    }

    private ResponseStatusException badRequest(String message) {

        return new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                message
        );
    }
}

            onClick={() => setActiveView("INVENTORY")}
            className="manager-link-button"
          >
            Inventory
          </button>

          <button
            onClick={() => setActiveView("RECIPES")}
            className="manager-link-button"
          >
            Recipes
          </button>

          <a href="/admin/qr" className="manager-link-button">
            QR Management
          </a>

          <button
            onClick={logout}
            className="manager-logout-button"
          >
            Logout
          </button>

        </div>

      </header>

      {error && (
        <div className="manager-error manager-dashboard-error">
          {error}
        </div>
      )}

      {activeView === "ANALYTICS" ? (
        <SalesAnalytics adminKey={adminKey} />
      ) : activeView === "INVENTORY" ? (
        <InventoryManagement adminKey={adminKey} />
      ) : activeView === "RECIPES" ? (
        <RecipeManagement adminKey={adminKey} />
      ) : (
        <>
      {/* STATISTICS */}

      <section className="manager-stats">

        <article className="manager-stat-card">
          <span>Total orders</span>
          <strong>{orders.length}</strong>
        </article>

        <article className="manager-stat-card pending-stat">
          <span>Pending orders</span>
          <strong>{pendingCount}</strong>
        </article>

        <article className="manager-stat-card preparing-stat">
          <span>Preparing</span>
          <strong>{preparingCount}</strong>
        </article>

        <article className="manager-stat-card ready-stat">
          <span>Ready to serve</span>
          <strong>{readyCount}</strong>
        </article>

        <article className="manager-stat-card revenue-stat">
          <span>Accepted order value</span>
          <strong>{formatMoney(totalRevenue)}</strong>
        </article>

      </section>

      {/* ORDER MANAGEMENT */}

      <section className="manager-orders-section">

        <div className="manager-orders-heading">

          <div>
            <span className="manager-eyebrow">
              LIVE RESTAURANT ORDERS
            </span>

            <h2>Customer orders</h2>
          </div>

          <div className="manager-last-updated">
            {lastUpdated
              ? `Updated ${lastUpdated.toLocaleTimeString()}`
              : "Waiting for orders"}
          </div>

        </div>

        {/* FILTERS */}

        <div className="manager-filters">

          {(
            [
              "ALL",
              "PENDING",
              "ACCEPTED",
              "PREPARING",
              "READY",
              "SERVED",
              "REJECTED",
              "CANCELLED",
            ] as OrderFilter[]
          ).map((status) => (

            <button
              key={status}
              className={
                filter === status
                  ? "manager-filter active"
                  : "manager-filter"
              }
              onClick={() => setFilter(status)}
            >
              {status === "ALL"
                ? "All orders"
                : status.replace("_", " ")}
            </button>

          ))}

        </div>

        {/* ORDER CARDS */}

        {filteredOrders.length === 0 ? (

          <div className="manager-empty-state">
            <span>🍽️</span>

            <h3>No orders found</h3>

            <p>
              {filter === "ALL"
                ? "New customer orders will appear here automatically."
                : `There are currently no ${filter.toLowerCase()} orders.`}
            </p>
          </div>

        ) : (

          <div className="manager-order-list">

            {filteredOrders.map((order) => {

              const nextAction = getNextAction(order.status);

              const isUpdating =
                updatingOrder === order.orderNumber;

              return (

                <article
                  className="manager-order-card"
                  key={order.id}
                >

                  {/* ORDER HEADER */}

                  <div className="manager-order-top">

                    <div>
                      <span className="manager-order-number">
                        {order.orderNumber}
                      </span>

                      <h3>
                        Table {order.tableNumber}
                      </h3>

                      <span className="manager-order-date">
                        {formatDate(order.createdAt)}
                      </span>
                    </div>

                    <span
                      className={`manager-status status-${order.status.toLowerCase()}`}
                    >
                      {order.status.replace("_", " ")}
                    </span>

                  </div>

                  {/* PAYMENT DETAILS */}

                  <div className="manager-payment-panel">

                    <div className="manager-payment-heading">
                      <strong>PAYMENT DETAILS</strong>

                      <span className={`manager-payment-badge payment-status-${order.paymentStatus.toLowerCase()}`}>
                        {order.paymentStatus === "CASH_DUE"
                          ? "CASH DUE"
                          : order.paymentStatus.replace("_", " ")}
                      </span>
                    </div>

                    <div className="manager-payment-grid">

                      <div>
                        <span>Payment method</span>
                        <strong>
                          {order.paymentMethod === "SSLCOMMERZ"
                            ? "SSLCommerz"
                            : "Cash"}
                        </strong>
                      </div>

                      <div>
                        <span>Payment amount</span>
                        <strong>
                          {order.paymentAmount != null
                            ? `${formatMoney(Number(order.paymentAmount))} ${order.paymentCurrency || "BDT"}`
                            : "Not received"}
                        </strong>
                      </div>

                      <div>
                        <span>Transaction ID</span>
                        <strong>
                          {order.transactionId || "Not applicable"}
                        </strong>
                      </div>

                      <div>
                        <span>Payment date</span>
                        <strong>
                          {order.paidAt
                            ? formatDate(order.paidAt)
                            : "Not paid yet"}
                        </strong>
                      </div>

                    </div>

                    {order.paymentMethod === "CASH" &&
                      order.paymentStatus === "CASH_DUE" && (

                      <button
                        className="manager-cash-paid-button"
                        disabled={isUpdating}
                        onClick={() => markCashAsPaid(order.orderNumber)}
                      >
                        {isUpdating
                          ? "Updating..."
                          : "Mark cash as paid"}
                      </button>

                    )}

                  </div>

                  {/* ORDER ITEMS */}

                  <div className="manager-items">

                    {order.items.map((item, index) => (

                      <div
                        className="manager-item"
                        key={`${item.foodId}-${index}`}
                      >

                        <div className="manager-item-heading">

                          <strong>
                            {item.foodName}
                          </strong>

                          <span>
                            × {item.quantity}
                          </span>

                        </div>

                        <div className="manager-item-details">

                          <span>
                            Spice: {item.spiceLevel}
                          </span>

                          <span>
                            Preparation: {item.preparationTimeMinutes} min
                          </span>

                        </div>

                        {(item.selectedAddons ?? []).length > 0 && (

                          <div className="manager-addons">

                            <span>Add-ons:</span>

                            {item.selectedAddons.map(
                              (addon, addonIndex) => (

                                <span key={addonIndex}>
                                  {addon.name} (+{formatMoney(addon.price)})
                                </span>

                              )
                            )}

                          </div>

                        )}

                        <div className="manager-item-price">
                          {formatMoney(item.lineTotal)}
                        </div>

                      </div>

                    ))}

                  </div>

                  {/* CUSTOMER NOTE */}

                  {order.customerNote && (

                    <div className="manager-customer-note">
                      <strong>Customer note:</strong>
                      <p>{order.customerNote}</p>
                    </div>

                  )}

                  {/* ORDER TOTAL */}

                  <div className="manager-order-summary">

                    <div>
                      <span>Estimated preparation time</span>
                      <strong>
                        {order.totalPreparationWorkMinutes} min
                      </strong>
                    </div>

                    <div>
                      <span>Order total</span>
                      <strong>
                        {formatMoney(order.totalAmount)}
                      </strong>
                    </div>

                  </div>

                  {/* ORDER ACTIONS */}

                  {nextAction && (

                    <div className="manager-order-actions">

                      {order.status === "PENDING" && (

                        <button
                          className="manager-reject-button"
                          disabled={isUpdating}
                          onClick={() =>
                            rejectOrder(order.orderNumber)
                          }
                        >
                          Reject
                        </button>

                      )}

                      {(order.status === "ACCEPTED" ||
                        order.status === "PREPARING") && (
                        <button
                          className="manager-reject-button"
                          disabled={isUpdating}
                          onClick={() => cancelOrder(order.orderNumber)}
                        >
                          Cancel order
                        </button>
                      )}

                      <button
                        className="manager-primary-button"
                        disabled={isUpdating}
                        onClick={() =>
                          updateStatus(
                            order.orderNumber,
                            nextAction.nextStatus
                          )
                        }
                      >
                        {isUpdating
                          ? "Updating..."
                          : nextAction.label}
                      </button>

                    </div>

                  )}

                </article>

              );

            })}

          </div>

        )}

      </section>

        </>
      )}

      <footer className="manager-footer">
        Khai-Dai Restaurant Management System · Manager Dashboard
      </footer>

    </main>
  );
}
