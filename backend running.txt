    $pg = $pgServices | Select-Object -First 1

}

    throw "PostgreSQL is not reachable on port 5432."
}

# 2. SECURE CREDENTIAL INPUT
# ==========================================

function Set-ProcessSecret {
    param(
        [string]$Name,
        [string]$Prompt
    )

    $secure = Read-Host $Prompt -AsSecureString
    $ptr = [Runtime.InteropServices.Marshal]::SecureStringToBSTR($secure)

    try {
        [Environment]::SetEnvironmentVariable(
            $Name,
            [Runtime.InteropServices.Marshal]::PtrToStringBSTR($ptr),
            "Process"
        )
    }
    finally {
        [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($ptr)
    }
}

# 7. NGROK
# ==========================================

Write-Host "`n===== 7. NGROK TUNNEL =====" -ForegroundColor Cyan

$ngrokCommand = Get-Command ngrok -ErrorAction SilentlyContinue

if (!$ngrokCommand) {
    throw "ngrok command not found. Install ngrok or configure its PATH."
}

$tunnel = $null

# Reuse existing tunnel if available
try {
    $existing = Invoke-RestMethod "http://127.0.0.1:4040/api/tunnels" -TimeoutSec 3

    $tunnel = $existing.tunnels |
        Where-Object {
            $_.public_url -like "https://*" -and
            $_.config.addr -match "8090"
        } |
        Select-Object -First 1
}
catch {
    $tunnel = $null
}

# Start ngrok if necessary
if (!$tunnel) {
    Write-Host "Starting ngrok..." -ForegroundColor Yellow

    Start-Process -FilePath $ngrokCommand.Source `
        -ArgumentList "http", "8090"

    for ($i = 0; $i -lt 30; $i++) {
        Start-Sleep -Seconds 1

        try {
            $existing = Invoke-RestMethod "http://127.0.0.1:4040/api/tunnels" -TimeoutSec 2

            $tunnel = $existing.tunnels |
                Where-Object {
                    $_.public_url -like "https://*" -and
                    $_.config.addr -match "8090"
                } |
                Select-Object -First 1

            if ($tunnel) {
                break
            }
        }
        catch {
            # Wait for ngrok startup
        }
    }
}

if (!$tunnel) {
    throw "Could not establish ngrok tunnel. Check ngrok window."
}

$env:KHAIDAI_PUBLIC_BACKEND_URL = $tunnel.public_url.TrimEnd('/')

Write-Host "Public backend: $env:KHAIDAI_PUBLIC_BACKEND_URL" -ForegroundColor Green
