package com.khaidai.backend.controller;

import com.khaidai.backend.dto.PaymentInitiateRequest;
import com.khaidai.backend.dto.PaymentInitiateResponse;
import com.khaidai.backend.service.PaymentService;

import jakarta.validation.Valid;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.util.MultiValueMap;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.util.UriComponentsBuilder;

import java.net.URI;
import java.util.Locale;

@RestController
@RequestMapping("/api/payments")

public class PaymentController {

    private final PaymentService paymentService;

    @Value("${sslcommerz.frontend-base-url:http://localhost:5173}")
    private String frontendBaseUrl;

    public PaymentController(PaymentService paymentService) {
        this.paymentService = paymentService;
    }

@CrossOrigin(
    originPatterns = {
        "http://localhost:*",
        "http://127.0.0.1:*",
        "http://192.168.*:*",
        "http://10.*:*",
        "http://172.*:*"
    }
)
@PostMapping("/initiate")
public ResponseEntity<PaymentInitiateResponse> initiatePayment(
        @Valid @RequestBody PaymentInitiateRequest request
) {
    return ResponseEntity.ok(
            paymentService.initiatePayment(request)
    );
}

    @PostMapping("/callback/success")
    public ResponseEntity<Void> success(
            @RequestParam MultiValueMap<String, String> params
    ) {
        return redirect(
                paymentService.handleGatewayCallback(params, "success")
        );
    }

    @PostMapping("/callback/fail")
    public ResponseEntity<Void> fail(
            @RequestParam MultiValueMap<String, String> params
    ) {
        return redirect(
                paymentService.handleGatewayCallback(params, "fail")
        );
    }

    @PostMapping("/callback/cancel")
    public ResponseEntity<Void> cancel(
            @RequestParam MultiValueMap<String, String> params
    ) {
        return redirect(
                paymentService.handleGatewayCallback(params, "cancel")
        );
    }

    @PostMapping("/callback/ipn")
    public ResponseEntity<String> ipn(
            @RequestParam MultiValueMap<String, String> params
    ) {
        paymentService.handleGatewayCallback(params, "ipn");

        return ResponseEntity.ok("IPN received");
    }

    private ResponseEntity<Void> redirect(
            PaymentService.CallbackResult result
    ) {
        URI destination = UriComponentsBuilder
                .fromUriString(frontendBaseUrl)
                .queryParam(
                        "paymentResult",
                        result.paymentStatus().name().toLowerCase(Locale.ROOT)
                )
                .queryParam("orderNumber", result.orderNumber())
                .build()
                .encode()
                .toUri();

        return ResponseEntity
                .status(HttpStatus.SEE_OTHER)
                .location(destination)
                .build();
    }
}

    @JoinColumn(name = "order_id", nullable = false)
    private RestaurantOrder order;

    @Column(name = "food_id", nullable = false)
    private Integer foodId;

    @Column(name = "food_name", nullable = false, length = 120)
    private String foodName;

    @Column(nullable = false)
    private Integer quantity;

    @Column(name = "unit_price", nullable = false, precision = 10, scale = 2)
    private BigDecimal unitPrice;

    @Column(name = "spice_level", nullable = false, length = 30)
    private String spiceLevel;

    @Column(name = "preparation_time_minutes", nullable = false)
    private Integer preparationTimeMinutes;

    @ElementCollection
    @CollectionTable(
        name = "order_item_addons",
        joinColumns = @JoinColumn(name = "order_item_id")
    )
    private List<OrderItemAddon> selectedAddons = new ArrayList<>();

    public OrderItem() {
    }

    public OrderItem(
            Integer foodId,
            String foodName,
            Integer quantity,
            BigDecimal unitPrice,
            String spiceLevel,
            Integer preparationTimeMinutes,
            List<OrderItemAddon> selectedAddons
    ) {
        this.foodId = foodId;
        this.foodName = foodName;
        this.quantity = quantity;
        this.unitPrice = unitPrice;
        this.spiceLevel = spiceLevel;
        this.preparationTimeMinutes = preparationTimeMinutes;

        if (selectedAddons != null) {
            this.selectedAddons = new ArrayList<>(selectedAddons);
        }
    }

    public BigDecimal getLineTotal() {

        BigDecimal addonTotal = selectedAddons.stream()
                .map(OrderItemAddon::getPrice)
                .reduce(BigDecimal.ZERO, BigDecimal::add);

        return unitPrice.add(addonTotal)
                .multiply(BigDecimal.valueOf(quantity));
    }

        return id;
    }

    public RestaurantOrder getOrder() {
        return order;
    }

    public void setOrder(RestaurantOrder order) {
        this.order = order;
    }

    public Integer getFoodId() {
        return foodId;
    }

    public String getFoodName() {
        return foodName;
    }

    public Integer getQuantity() {
        return quantity;
    }

    public BigDecimal getUnitPrice() {
        return unitPrice;
    }

    public String getSpiceLevel() {
        return spiceLevel;
    }

    public Integer getPreparationTimeMinutes() {
        return preparationTimeMinutes;
    }

package com.khaidai.backend.service;

import com.khaidai.backend.dto.AdminAnalyticsResponse;
import com.khaidai.backend.entity.*;
import com.khaidai.backend.repository.OrderRepository;
import com.khaidai.backend.repository.PaymentRepository;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;

import java.math.BigDecimal;
import java.math.RoundingMode;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;

import java.time.Instant;
import java.time.LocalDate;
import java.time.ZoneId;
import java.time.temporal.ChronoUnit;

import java.util.*;

@Service
public class AdminAnalyticsService {

    private static final ZoneId BUSINESS_ZONE =
            ZoneId.of("Asia/Dhaka");

    private static final BigDecimal ZERO =
            new BigDecimal("0.00");

    private final OrderRepository orderRepository;
    private final PaymentRepository paymentRepository;

    @Value("${khaidai.admin.api-key:}")
    private String configuredAdminKey;

    public AdminAnalyticsService(
            OrderRepository orderRepository,
            PaymentRepository paymentRepository
    ) {
        this.orderRepository = orderRepository;
        this.paymentRepository = paymentRepository;
    }

    @Transactional(readOnly = true)
    public AdminAnalyticsResponse getAnalytics(
            String adminKey,
            LocalDate fromDate,
            LocalDate toDate
    ) {

        validateAdminKey(adminKey);

        if (fromDate == null || toDate == null) {
            throw badRequest("Both start and end dates are required.");
        }

        if (fromDate.isAfter(toDate)) {
            throw badRequest("Start date cannot be after end date.");
        }

        long days = ChronoUnit.DAYS.between(fromDate, toDate) + 1;

        if (days > 90) {
            throw badRequest("Analytics date range cannot exceed 90 days.");
        }

        Instant fromInstant =
                fromDate.atStartOfDay(BUSINESS_ZONE).toInstant();

        Instant toInstant =
                toDate.plusDays(1).atStartOfDay(BUSINESS_ZONE).toInstant();

        // Orders are counted by order creation date.
        List<RestaurantOrder> orders =
                orderRepository
                        .findAllByCreatedAtGreaterThanEqualAndCreatedAtLessThanOrderByCreatedAtAsc(
                                fromInstant,
                                toInstant
                        );

        // Revenue is counted by verified payment date.
        List<Payment> payments =
                paymentRepository
                        .findAllByStatusAndPaidAtGreaterThanEqualAndPaidAtLessThanOrderByPaidAtAsc(
                                PaymentStatus.PAID,
                                fromInstant,
                                toInstant
                        );

        BigDecimal cashRevenue = BigDecimal.ZERO;
        BigDecimal onlineRevenue = BigDecimal.ZERO;

        long cashPayments = 0;
        long onlinePayments = 0;

        Set<Long> paidOrderIds = new HashSet<>();

        Map<LocalDate, BigDecimal> dailyTotals =
                new LinkedHashMap<>();

        for (
                LocalDate date = fromDate;
                !date.isAfter(toDate);
                date = date.plusDays(1)
        ) {
            dailyTotals.put(date, BigDecimal.ZERO);
        }

        Map<Integer, ItemAccumulator> itemTotals =
                new HashMap<>();

        for (Payment payment : payments) {

            if (!"BDT".equalsIgnoreCase(payment.getCurrency())) {
                continue;
            }

            BigDecimal amount = payment.getAmount() == null
                    ? BigDecimal.ZERO
                    : payment.getAmount();

            LocalDate paymentDate =
                    payment.getPaidAt()
                            .atZone(BUSINESS_ZONE)
                            .toLocalDate();

            dailyTotals.compute(
                    paymentDate,
                    (date, total) ->
                            (total == null ? BigDecimal.ZERO : total)
                                    .add(amount)
            );

            if (payment.getPaymentMethod() == PaymentMethod.CASH) {

                cashRevenue = cashRevenue.add(amount);
                cashPayments++;

            } else if (
                    payment.getPaymentMethod() == PaymentMethod.SSLCOMMERZ
            ) {

                onlineRevenue = onlineRevenue.add(amount);
                onlinePayments++;
            }

            // Count each paid order only once when calculating
            // sold food quantities, even if multiple payments exist.
            RestaurantOrder order = payment.getOrder();

            if (paidOrderIds.add(order.getId())) {

                for (OrderItem item : order.getItems()) {

                    ItemAccumulator accumulator =
                            itemTotals.computeIfAbsent(
                                    item.getFoodId(),
                                    id -> new ItemAccumulator(
                                            item.getFoodId(),
                                            item.getFoodName()
                                    )
                            );

                    accumulator.quantity += item.getQuantity();

                    accumulator.sales =
                            accumulator.sales.add(item.getLineTotal());
                }
            }
        }

        BigDecimal totalRevenue =
                cashRevenue.add(onlineRevenue);

        BigDecimal averageOrderValue = paidOrderIds.isEmpty()
                ? ZERO
                : totalRevenue.divide(
                        BigDecimal.valueOf(paidOrderIds.size()),
                        2,
                        RoundingMode.HALF_UP
                );

        // Order status counts by creation date.
        Map<OrderStatus, Long> statusTotals =
                new EnumMap<>(OrderStatus.class);

        for (OrderStatus status : OrderStatus.values()) {
            statusTotals.put(status, 0L);
        }

        for (RestaurantOrder order : orders) {
            statusTotals.merge(order.getStatus(), 1L, Long::sum);
        }

        List<AdminAnalyticsResponse.OrderStatusCount> statusCounts =
                new ArrayList<>();

        for (OrderStatus status : OrderStatus.values()) {

            statusCounts.add(
                    new AdminAnalyticsResponse.OrderStatusCount(
                            status.name(),
                            statusTotals.get(status)
                    )
            );
        }

        List<AdminAnalyticsResponse.DailyRevenue> dailyRevenue =
                dailyTotals.entrySet()
                        .stream()
                        .map(entry ->
                                new AdminAnalyticsResponse.DailyRevenue(
                                        entry.getKey(),
                                        money(entry.getValue())
                                )
                        )
                        .toList();

        List<ItemAccumulator> sortedItems =
                new ArrayList<>(itemTotals.values());

        sortedItems.sort((left, right) -> {

            int quantityComparison =
                    Long.compare(right.quantity, left.quantity);

            if (quantityComparison != 0) {
                return quantityComparison;
            }

            return left.foodName.compareToIgnoreCase(right.foodName);
        });

        List<AdminAnalyticsResponse.TopSellingItem> topSellingItems =
                sortedItems.stream()
                        .limit(5)
                        .map(item ->
                                new AdminAnalyticsResponse.TopSellingItem(
                                        item.foodId,
                                        item.foodName,
                                        item.quantity,
                                        money(item.sales)
                                )
                        )
                        .toList();

        return new AdminAnalyticsResponse(

                fromDate,
                toDate,

                money(totalRevenue),

                orders.size(),
                paidOrderIds.size(),

                money(averageOrderValue),

                money(cashRevenue),
                money(onlineRevenue),

                cashPayments,
                onlinePayments,

                dailyRevenue,
                topSellingItems,
                statusCounts
        );
    }

    private void validateAdminKey(String providedKey) {

        if (configuredAdminKey == null ||
                configuredAdminKey.isBlank()) {

            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "Admin API key is not configured."
            );
        }

        if (providedKey == null ||
                !MessageDigest.isEqual(
                        configuredAdminKey.getBytes(StandardCharsets.UTF_8),
                        providedKey.getBytes(StandardCharsets.UTF_8)
                )) {

            throw new ResponseStatusException(
                    HttpStatus.UNAUTHORIZED,
                    "Invalid administrator key."
            );
        }
    }

    private BigDecimal money(BigDecimal value) {
        return value.setScale(2, RoundingMode.HALF_UP);
    }

    private ResponseStatusException badRequest(String message) {

        return new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                message
        );
    }

    private static final class ItemAccumulator {

        private final Integer foodId;
        private final String foodName;

        private long quantity = 0;
        private BigDecimal sales = BigDecimal.ZERO;

        private ItemAccumulator(Integer foodId, String foodName) {
            this.foodId = foodId;
            this.foodName = foodName;
        }
    }
}

package com.khaidai.backend.service;

import com.khaidai.backend.dto.InventoryOverviewResponse;
import com.khaidai.backend.dto.InventoryRequests;

import com.khaidai.backend.repository.InventoryIngredientRepository;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;

import java.math.BigDecimal;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;

import java.util.List;
import java.util.Locale;
import java.util.Set;

@Service
public class InventoryService {

    private static final Set<String> ALLOWED_UNITS = Set.of(
            "g", "kg", "ml", "L", "pcs", "pack", "bottle", "box"
    );

    private final InventoryIngredientRepository ingredientRepository;
    private final StockMovementRepository movementRepository;

    @Value("${khaidai.admin.api-key:}")
    private String configuredAdminKey;

    public InventoryService(
            InventoryIngredientRepository ingredientRepository,
            StockMovementRepository movementRepository
    ) {
        this.ingredientRepository = ingredientRepository;
        this.movementRepository = movementRepository;
    }

    // ==========================================
    // GET INVENTORY OVERVIEW
    // ==========================================

    @Transactional(readOnly = true)
    public InventoryOverviewResponse getOverview(String adminKey) {

        validateAdminKey(adminKey);

        List<InventoryOverviewResponse.IngredientRow> ingredients =
                ingredientRepository.findAllByOrderByNameAsc()
                        .stream()
                        .map(this::toIngredientRow)
                        .toList();

        List<InventoryOverviewResponse.MovementRow> movements =
                movementRepository.findTop50ByOrderByCreatedAtDesc()
                        .stream()
                        .map(this::toMovementRow)
                        .toList();

        return new InventoryOverviewResponse(
                ingredients,
                movements
        );
    }

    // ==========================================
    // CREATE INGREDIENT
    // ==========================================

    @Transactional
    public void createIngredient(
            InventoryRequests.CreateIngredientRequest request,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        if (request == null) {
            throw badRequest("Ingredient information is required.");
        }

        String name = cleanName(request.name());
        String unit = validateUnit(request.unit());

        BigDecimal openingStock = nonNegative(
                request.openingStock(),
                "Opening stock"
        );

        BigDecimal reorderLevel = nonNegative(
                request.reorderLevel(),
                "Reorder level"
        );

        if (ingredientRepository.findByNameIgnoreCase(name).isPresent()) {
            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "An ingredient with this name already exists."
            );
        }

        InventoryIngredient ingredient = new InventoryIngredient();

        ingredient.setName(name);
        ingredient.setUnit(unit);
        ingredient.setCurrentStock(openingStock);
        ingredient.setReorderLevel(reorderLevel);
        ingredient.setActive(true);

        ingredientRepository.saveAndFlush(ingredient);

        if (openingStock.compareTo(BigDecimal.ZERO) > 0) {

            movementRepository.save(
                    new StockMovement(
                            ingredient,
                            StockMovementType.OPENING,
                            openingStock,
                            openingStock,
                            "Opening stock"
                    )
            );
        }
    }

    // ==========================================
    // UPDATE INGREDIENT
    // ==========================================

    @Transactional
    public void updateIngredient(
            Long id,
            InventoryRequests.UpdateIngredientRequest request,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        if (request == null) {
            throw badRequest("Update information is required.");
        }

        InventoryIngredient ingredient = findForUpdate(id);

        if (request.name() != null) {

            String name = cleanName(request.name());

            ingredientRepository.findByNameIgnoreCase(name)
                    .filter(existing -> !existing.getId().equals(id))
                    .ifPresent(existing -> {
                        throw new ResponseStatusException(
                                HttpStatus.CONFLICT,
                                "Another ingredient already uses this name."
                        );
                    });

            ingredient.setName(name);
        }

        if (request.unit() != null) {
            ingredient.setUnit(validateUnit(request.unit()));
        }

        if (request.reorderLevel() != null) {
            ingredient.setReorderLevel(
                    nonNegative(
                            request.reorderLevel(),
                            "Reorder level"
                    )
            );
        }

        if (request.active() != null) {
            ingredient.setActive(request.active());
        }

        ingredientRepository.save(ingredient);
    }

    // ==========================================
    // RECORD STOCK MOVEMENT
    // ==========================================

    @Transactional
    public void recordMovement(
            Long id,
            InventoryRequests.StockMovementRequest request,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        if (request == null || request.type() == null) {
            throw badRequest("Stock movement information is required.");
        }

        InventoryIngredient ingredient = findForUpdate(id);

        if (!ingredient.isActive()) {
            throw badRequest("This ingredient is inactive.");
        }

        StockMovementType type;

        try {
            type = StockMovementType.valueOf(
                    request.type().trim().toUpperCase(Locale.ROOT)
            );
        } catch (IllegalArgumentException exception) {
            throw badRequest("Invalid stock movement type.");
        }

        if (type == StockMovementType.OPENING) {
            throw badRequest("Opening stock is only allowed when creating an ingredient.");
        }

        BigDecimal quantity = nonNegative(
                request.quantity(),
                "Quantity"
        );

        if (quantity.compareTo(BigDecimal.ZERO) == 0 &&
                type != StockMovementType.ADJUSTMENT) {
            throw badRequest("Quantity must be greater than zero.");
        }

        BigDecimal current = ingredient.getCurrentStock();

        BigDecimal change;
        BigDecimal newBalance;

        switch (type) {

            case STOCK_IN -> {

                change = quantity;
                newBalance = current.add(quantity);
            }

            case WASTE -> {

                change = quantity.negate();
                newBalance = current.subtract(quantity);

                if (newBalance.compareTo(BigDecimal.ZERO) < 0) {
                    throw new ResponseStatusException(
                            HttpStatus.CONFLICT,
                            "Waste quantity exceeds available stock."
                    );
                }
            }

            case ADJUSTMENT -> {

                // For adjustment, quantity means actual
                // physically counted stock, not a delta.

                newBalance = quantity;
                change = newBalance.subtract(current);
            }

            default -> throw badRequest("Unsupported stock movement.");
        }

        ingredient.setCurrentStock(newBalance);

        ingredientRepository.save(ingredient);

        String note = request.note() == null
                ? null
                : request.note().trim();

        if (note != null && note.isEmpty()) {
            note = null;
        }

        if (note != null && note.length() > 250) {
            throw badRequest("Movement note cannot exceed 250 characters.");
        }

        movementRepository.save(
                new StockMovement(
                        ingredient,
                        type,
                        change,
                        newBalance,
                        note
                )
        );
    }

    // ==========================================
    // MAPPING
    // ==========================================

    private InventoryOverviewResponse.IngredientRow toIngredientRow(
            InventoryIngredient ingredient
    ) {

        boolean lowStock =
                ingredient.getCurrentStock()
                        .compareTo(ingredient.getReorderLevel()) <= 0;

        return new InventoryOverviewResponse.IngredientRow(

                ingredient.getId(),
                ingredient.getName(),
                ingredient.getUnit(),

                ingredient.getCurrentStock(),
                ingredient.getReorderLevel(),

                ingredient.isActive(),
                lowStock,

                ingredient.getUpdatedAt()
        );
    }

    private InventoryOverviewResponse.MovementRow toMovementRow(
            StockMovement movement
    ) {

        InventoryIngredient ingredient = movement.getIngredient();

        return new InventoryOverviewResponse.MovementRow(

                movement.getId(),
                ingredient.getId(),

                ingredient.getName(),
                ingredient.getUnit(),

                movement.getMovementType().name(),

                movement.getQuantityChange(),
                movement.getBalanceAfter(),

                movement.getNote(),
                movement.getCreatedAt()
        );
    }

    // ==========================================
    // VALIDATION
    // ==========================================

    private InventoryIngredient findForUpdate(Long id) {

        if (id == null) {
            throw badRequest("Ingredient ID is required.");
        }

        return ingredientRepository.findByIdForUpdate(id)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Ingredient not found."
                ));
    }

    private String cleanName(String value) {

        if (value == null || value.isBlank()) {
            throw badRequest("Ingredient name is required.");
        }

        String name = value.trim();

        if (name.length() > 100) {
            throw badRequest("Ingredient name cannot exceed 100 characters.");
        }

        return name;
    }

    private String validateUnit(String value) {

        if (value == null || !ALLOWED_UNITS.contains(value.trim())) {
            throw badRequest(
                    "Unit must be g, kg, ml, L, pcs, pack, bottle or box."
            );
        }

        return value.trim();
    }

    private BigDecimal nonNegative(
            BigDecimal value,
            String field
    ) {

        if (value == null || value.compareTo(BigDecimal.ZERO) < 0) {
            throw badRequest(field + " cannot be negative.");
        }

        return value;
    }

    private void validateAdminKey(String providedKey) {

        if (configuredAdminKey == null ||
                configuredAdminKey.isBlank()) {

            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "Admin API key is not configured."
            );
        }

        if (providedKey == null ||
                !MessageDigest.isEqual(
                        configuredAdminKey.getBytes(StandardCharsets.UTF_8),
                        providedKey.getBytes(StandardCharsets.UTF_8)
                )) {

            throw new ResponseStatusException(
                    HttpStatus.UNAUTHORIZED,
                    "Invalid administrator key."
            );
        }
    }

    private ResponseStatusException badRequest(String message) {

        return new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                message
        );
    }
}

import org.springframework.stereotype.Service;

import org.springframework.web.server.ResponseStatusException;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.util.List;
import java.util.Locale;
import java.util.UUID;

@Service
public class OrderAdminService {

    private final OrderRepository orderRepository;
    private final PaymentRepository paymentRepository;

    @Value("${khaidai.admin.api-key:}")
    private String configuredAdminKey;

    public OrderAdminService(
            OrderRepository orderRepository,
            PaymentRepository paymentRepository
    ) {
        this.orderRepository = orderRepository;
        this.paymentRepository = paymentRepository;
    }

    @Transactional(readOnly = true)
    public List<AdminOrderResponse> getAllOrders(String adminKey) {

        validateAdminKey(adminKey);

        return orderRepository
                .findAllByOrderByCreatedAtDesc()
                .stream()
                .map(this::toResponse)
                .toList();
    }

    @Transactional
    public AdminOrderResponse updateOrderStatus(
            String orderNumber,
            String requestedStatus,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        if (requestedStatus == null || requestedStatus.isBlank()) {
            throw badRequest("Order status is required.");
        }

        OrderStatus newStatus;

        try {
            newStatus = OrderStatus.valueOf(
                    requestedStatus.trim().toUpperCase(Locale.ROOT)
            );
        } catch (IllegalArgumentException exception) {
            throw badRequest("Invalid order status.");
        }

        RestaurantOrder order = findOrder(orderNumber);

        if (!isValidTransition(order.getStatus(), newStatus)) {
            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "Cannot change order status from "
                            + order.getStatus() + " to " + newStatus + "."
            );
        }

        order.setStatus(newStatus);

        RestaurantOrder savedOrder = orderRepository.save(order);

        return toResponse(savedOrder);
    }

    // ==========================================
    // MARK CASH PAYMENT AS PAID
    // ==========================================

    @Transactional
    public AdminOrderResponse markCashPaymentAsPaid(
            String orderNumber,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        RestaurantOrder order = findOrder(orderNumber);

        if (order.getStatus() == OrderStatus.AWAITING_PAYMENT) {
            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "This order is awaiting online payment."
            );
        }

        List<Payment> existingPayments =
                paymentRepository.findAllByOrder_IdOrderByCreatedAtDesc(
                        order.getId()
                );

        Payment latest = existingPayments.isEmpty()
                ? null
                : existingPayments.get(0);

        if (latest != null &&
                latest.getStatus() == PaymentStatus.PAID) {

            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "This order is already marked as paid."
            );
        }

        if (latest != null &&
                latest.getPaymentMethod() == PaymentMethod.SSLCOMMERZ &&
                latest.getStatus() == PaymentStatus.PENDING) {

            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "An online payment is still pending."
            );
        }

        Payment cashPayment = new Payment();

        cashPayment.setOrder(order);

        cashPayment.setTransactionId(
                "CASH-" + UUID.randomUUID()
                        .toString()
                        .replace("-", "")
                        .substring(0, 24)
                        .toUpperCase(Locale.ROOT)
        );

        cashPayment.setPaymentMethod(PaymentMethod.CASH);
        cashPayment.setStatus(PaymentStatus.PAID);
        cashPayment.setAmount(order.getTotalAmount());
        cashPayment.setCurrency("BDT");

        paymentRepository.saveAndFlush(cashPayment);

        return toResponse(order);
    }

    private RestaurantOrder findOrder(String orderNumber) {

        return orderRepository
                .findByOrderNumber(orderNumber)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Order not found."
                ));
    }

    private boolean isValidTransition(
            OrderStatus current,
            OrderStatus next
    ) {

        return switch (current) {

            case PENDING ->
                    next == OrderStatus.ACCEPTED ||
                    next == OrderStatus.REJECTED;

            case ACCEPTED ->
                    next == OrderStatus.PREPARING;

            case PREPARING ->
                    next == OrderStatus.READY;

            case READY ->
                    next == OrderStatus.SERVED;

            default -> false;
        };
    }

    private void validateAdminKey(String providedAdminKey) {

        if (configuredAdminKey == null ||
                configuredAdminKey.isBlank()) {

            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "Admin API key is not configured."
            );
        }

        if (providedAdminKey == null ||
                !MessageDigest.isEqual(
                        configuredAdminKey.getBytes(StandardCharsets.UTF_8),
                        providedAdminKey.getBytes(StandardCharsets.UTF_8)
                )) {

            throw new ResponseStatusException(
                    HttpStatus.UNAUTHORIZED,
                    "Invalid admin API key."
            );
        }
    }

    // ==========================================
    // MAP ORDER AND LATEST PAYMENT
    // ==========================================

    private AdminOrderResponse toResponse(RestaurantOrder order) {

        List<Payment> payments =
                paymentRepository.findAllByOrder_IdOrderByCreatedAtDesc(
                        order.getId()
                );

        Payment latest = payments.isEmpty()
                ? null
                : payments.get(0);

        String paymentMethod = latest != null
                ? latest.getPaymentMethod().name()
                : order.getStatus() == OrderStatus.AWAITING_PAYMENT
                    ? "SSLCOMMERZ"
                    : "CASH";

        String paymentStatus = latest != null
                ? latest.getStatus().name()
                : "CASH".equals(paymentMethod)
                    ? "CASH_DUE"
                    : "PENDING";

        List<AdminOrderResponse.ItemResponse> items =
                order.getItems()
                        .stream()
                        .map(item ->
                                new AdminOrderResponse.ItemResponse(
                                        item.getFoodId(),
                                        item.getFoodName(),
                                        item.getQuantity(),
                                        item.getUnitPrice(),
                                        item.getLineTotal(),
                                        item.getSpiceLevel(),
                                        item.getPreparationTimeMinutes(),
                                        item.getSelectedAddons()
                                                .stream()
                                                .map(addon ->
                                                        new AdminOrderResponse.AddonResponse(
                                                                addon.getName(),
                                                                addon.getPrice()
                                                        )
                                                )
                                                .toList()
                                )
                        )
                        .toList();

        return new AdminOrderResponse(
                order.getId(),
                order.getOrderNumber(),
                order.getRestaurantTable().getTableNumber(),
                order.getStatus(),
                order.getTotalAmount(),
                order.getTotalPreparationWorkMinutes(),
                order.getCustomerNote(),
                order.getCreatedAt(),
                order.getUpdatedAt(),

                paymentMethod,
                paymentStatus,
                latest != null ? latest.getTransactionId() : null,
                latest != null ? latest.getAmount() : null,
                latest != null ? latest.getCurrency() : "BDT",
                latest != null ? latest.getPaidAt() : null,

                items
        );
    }

    private ResponseStatusException badRequest(String message) {

        return new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                message
        );
    }
}

@Service
public class OrderService {

        this.tableRepository = tableRepository;

    // ==========================================
    // CREATE CUSTOMER ORDER
    // ==========================================

    @Transactional
    public OrderResponse createOrder(CreateOrderRequest request) {

        if (request == null) {
            throw badRequest("Order information is required.");
        }

        // Validate table QR token

        if (request.tableToken() == null ||
                request.tableToken().isBlank()) {

            throw badRequest("Table QR token is required.");
        }

        RestaurantTable table = tableRepository
                .findByQrTokenAndActiveTrue(request.tableToken())
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Invalid or inactive restaurant table."
                ));

        // Validate customer note

        if (request.customerNote() != null &&
                request.customerNote().length() > 500) {

            throw badRequest(
                    "Customer note cannot exceed 500 characters."
            );
        }

        // Validate order items

        if (request.items() == null ||
                request.items().isEmpty()) {

            throw badRequest(
                    "An order must contain at least one item."
            );
        }

        if (request.items().size() > 50) {

            throw badRequest(
                    "An order cannot contain more than 50 item lines."
            );
        }

        RestaurantOrder order = new RestaurantOrder(
                "KD-" + UUID.randomUUID().toString().toUpperCase(),
                table,
                request.customerNote()
        );

        BigDecimal orderTotal = BigDecimal.ZERO;

        int totalPreparationWork = 0;

        // Process every order item

        for (CreateOrderRequest.ItemRequest item : request.items()) {

            // Validate food and retrieve official catalog data

            MenuCatalog.FoodItem food = validateItem(item);

            List<OrderItemAddon> addons = new ArrayList<>();

            // Validate selected add-ons

            if (item.selectedAddons() != null) {

                if (item.selectedAddons().size() > 20) {

                    throw badRequest("Too many add-ons selected.");
                }

                // Prevent duplicate add-on selections

                Set<String> selectedAddonIds = new HashSet<>();

                for (CreateOrderRequest.AddonRequest addonRequest
                        : item.selectedAddons()) {

                    if (addonRequest == null ||
                            addonRequest.id() == null ||
                            addonRequest.id().isBlank()) {

                        throw badRequest(
                                "Invalid add-on information."
                        );
                    }

                    String addonId = addonRequest.id().trim();

                    if (!selectedAddonIds.add(addonId)) {

                        throw badRequest(
                                "Duplicate add-on selection."
                        );
                    }

                    // Find add-on belonging to this particular food

                    MenuCatalog.Addon officialAddon =
                            food.addons().get(addonId);

                    if (officialAddon == null) {

                        throw badRequest(
                                "Selected add-on is not available for this food."
                        );
                    }

                    // Use official backend add-on price

                    addons.add(new OrderItemAddon(
                            officialAddon.name(),
                            normalizePrice(officialAddon.price())
                    ));
                }
            }

            // Build order item using backend-authoritative values

            OrderItem orderItem = new OrderItem(

                    food.id(),

                    food.name(),

                    item.quantity(),

                    normalizePrice(food.price()),

                    item.spiceLevel(),

                    food.preparationTimeMinutes(),

                    addons
            );

            order.addItem(orderItem);

            // Calculate actual line total

            orderTotal = orderTotal.add(
                    orderItem.getLineTotal()
            );

            // Different dishes can be prepared simultaneously

            totalPreparationWork = Math.max(
                    totalPreparationWork,
                    food.preparationTimeMinutes()
            );
        }

        // Save calculated totals

        order.setTotalAmount(
                orderTotal.setScale(2, RoundingMode.HALF_UP)
        );

        order.setTotalPreparationWorkMinutes(
                totalPreparationWork
        );

        // Preserve existing behavior when paymentMethod is omitted.
        PaymentMethod paymentMethod = request.paymentMethod() == null
                ? PaymentMethod.CASH
                : request.paymentMethod();

        order.setStatus(
                paymentMethod == PaymentMethod.SSLCOMMERZ
                        ? OrderStatus.AWAITING_PAYMENT
                        : OrderStatus.PENDING
        );

        RestaurantOrder savedOrder = orderRepository.save(order);

        return toResponse(savedOrder);
    }

    // ==========================================
    // VALIDATE ITEM AND GET OFFICIAL FOOD
    // ==========================================

    private MenuCatalog.FoodItem validateItem(
            CreateOrderRequest.ItemRequest item
    ) {

        if (item == null) {

            throw badRequest("Invalid order item.");
        }

        if (item.foodId() == null) {

            throw badRequest("Invalid food ID.");
        }

        MenuCatalog.FoodItem food =
                MenuCatalog.findById(item.foodId());

        if (food == null) {

            throw badRequest("Food does not exist.");
        }

        if (item.quantity() == null ||
                item.quantity() < 1 ||
                item.quantity() > 50) {

            throw badRequest(
                    "Quantity must be between 1 and 50."
            );
        }

        // Validate spice level against this food's options

        if (item.spiceLevel() == null ||
                !food.spiceOptions().contains(item.spiceLevel())) {

            throw badRequest(
                    "Invalid spice level for this food."
            );
        }

        return food;
    }

    // ==========================================
    // NORMALIZE PRICE
    // ==========================================

    private BigDecimal normalizePrice(BigDecimal price) {

        return price.setScale(2, RoundingMode.HALF_UP);
    }

    // ==========================================
    // GET CUSTOMER ORDER
    // ==========================================

    @Transactional(readOnly = true)
    public OrderResponse getOrder(String orderNumber) {

        RestaurantOrder order = orderRepository
                .findByOrderNumber(orderNumber)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Order not found."
                ));

        return toResponse(order);
    }

    // ==========================================
    // CONVERT ENTITY TO RESPONSE
    // ==========================================

    private OrderResponse toResponse(RestaurantOrder order) {

        List<OrderResponse.ItemResponse> itemResponses =
                order.getItems().stream()

                        .map(item -> new OrderResponse.ItemResponse(

                                item.getFoodId(),

                                item.getFoodName(),

                                item.getQuantity(),

                                item.getUnitPrice(),

                                item.getLineTotal(),

                                item.getSpiceLevel(),

                                item.getPreparationTimeMinutes(),

                                item.getSelectedAddons().stream()

                                        .map(addon ->
                                                new OrderResponse.AddonResponse(
                                                        addon.getName(),
                                                        addon.getPrice()
                                                )
                                        )

                                        .toList()
                        ))

                        .toList();

        return new OrderResponse(

                order.getId(),

                order.getOrderNumber(),

                order.getRestaurantTable().getTableNumber(),

                order.getStatus(),

                order.getTotalAmount(),

                order.getTotalPreparationWorkMinutes(),

                order.getCreatedAt(),

                itemResponses
        );
    }

    // ==========================================
    // BAD REQUEST
    // ==========================================

    private ResponseStatusException badRequest(String message) {

        return new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                message
        );
    }
}

package com.khaidai.backend.service;

import org.springframework.http.HttpStatus;

@Service
public class PaymentService {

    private static final BigDecimal MIN_AMOUNT =
            new BigDecimal("10.00");

    private static final BigDecimal MAX_AMOUNT =
            new BigDecimal("500000.00");

    private final OrderRepository orderRepository;
    private final PaymentRepository paymentRepository;
    private final RestClient restClient = RestClient.create();

    @Value("${sslcommerz.store-id:}")
    private String storeId;

    @Value("${sslcommerz.store-password:}")
    private String storePassword;

    @Value("${sslcommerz.base-url:https://sandbox.sslcommerz.com}")
    private String baseUrl;

    @Value("${sslcommerz.callback-base-url:http://localhost:8090}")
    private String callbackBaseUrl;

    public PaymentService(
            OrderRepository orderRepository,
            PaymentRepository paymentRepository
    ) {
        this.orderRepository = orderRepository;
        this.paymentRepository = paymentRepository;
    }

    @Transactional(noRollbackFor = ResponseStatusException.class)
    public PaymentInitiateResponse initiatePayment(
            PaymentInitiateRequest request
    ) {

        if (!StringUtils.hasText(storeId) ||
                !StringUtils.hasText(storePassword)) {

            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "SSLCommerz credentials are not configured."
            );
        }

        RestaurantOrder order = orderRepository
                .findByOrderNumber(request.orderNumber())
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Order not found."
                ));

        if (order.getStatus() != OrderStatus.AWAITING_PAYMENT) {
            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "This order is not awaiting online payment."
            );
        }

        BigDecimal amount = order.getTotalAmount()
                .setScale(2, RoundingMode.HALF_UP);

        if (amount.compareTo(MIN_AMOUNT) < 0 ||
                amount.compareTo(MAX_AMOUNT) > 0) {

            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Payment amount must be between BDT 10 and BDT 500,000."
            );
        }

        // Reuse an existing pending session to avoid duplicate charges.
        List<Payment> previousPayments =
                paymentRepository.findAllByOrder_IdOrderByCreatedAtDesc(
                        order.getId()
                );

        for (Payment previous : previousPayments) {

            if (previous.getPaymentMethod() == PaymentMethod.SSLCOMMERZ &&
                    previous.getStatus() == PaymentStatus.PENDING) {

                if (StringUtils.hasText(previous.getGatewayPageUrl())) {
                    return toResponse(previous);
                }

                throw new ResponseStatusException(
                        HttpStatus.CONFLICT,
                        "A payment session is already being created. Please try again shortly."
                );
            }
        }

        String transactionId =
                "KD" + UUID.randomUUID()
                        .toString()
                        .replace("-", "")
                        .substring(0, 28)
                        .toUpperCase(Locale.ROOT);

        Payment payment = new Payment();

        payment.setOrder(order);
        payment.setTransactionId(transactionId);
        payment.setPaymentMethod(PaymentMethod.SSLCOMMERZ);
        payment.setStatus(PaymentStatus.PENDING);
        payment.setAmount(amount);
        payment.setCurrency("BDT");

        paymentRepository.saveAndFlush(payment);

        MultiValueMap<String, String> form =
                new LinkedMultiValueMap<>();

        form.add("store_id", storeId);
        form.add("store_passwd", storePassword);
        form.add("total_amount", amount.toPlainString());
        form.add("currency", "BDT");
        form.add("tran_id", transactionId);

        form.add("success_url", callbackUrl("success"));
        form.add("fail_url", callbackUrl("fail"));
        form.add("cancel_url", callbackUrl("cancel"));
        form.add("ipn_url", callbackUrl("ipn"));

        form.add("cus_name", request.customerName());
        form.add("cus_email", request.customerEmail());
        form.add("cus_phone", request.customerPhone());
        form.add("cus_add1", request.customerAddress());
        form.add("cus_city", request.customerCity());
        form.add("cus_country", "Bangladesh");

        if (StringUtils.hasText(request.customerPostcode())) {
            form.add("cus_postcode", request.customerPostcode());
        }

        int itemCount = order.getItems().stream()
                .mapToInt(item -> item.getQuantity())
                .sum();

        form.add("shipping_method", "NO");
        form.add("product_name", "Khai-Dai Restaurant Order");
        form.add("product_category", "food");
        form.add("product_profile", "general");
        form.add("num_of_item", String.valueOf(Math.max(itemCount, 1)));

        form.add("value_a", order.getOrderNumber());
        form.add("value_b", transactionId);

        Map<?, ?> gatewayResponse;

        try {

            String apiUrl = baseUrl.replaceAll("/+$", "")
                    + "/gwprocess/v4/api.php";

            gatewayResponse = restClient.post()
                    .uri(apiUrl)
                    .contentType(MediaType.APPLICATION_FORM_URLENCODED)
                    .accept(MediaType.APPLICATION_JSON)
                    .body(form)
                    .retrieve()
                    .body(Map.class);

        } catch (RestClientException exception) {

            payment.setStatus(PaymentStatus.FAILED);
            paymentRepository.save(payment);

            throw new ResponseStatusException(
                    HttpStatus.BAD_GATEWAY,
                    "Unable to connect to SSLCommerz. Please try again."
            );
        }

        if (gatewayResponse == null) {
            payment.setStatus(PaymentStatus.FAILED);
            paymentRepository.save(payment);

            throw new ResponseStatusException(
                    HttpStatus.BAD_GATEWAY,
                    "SSLCommerz returned an empty response."
            );
        }

        String gatewayStatus = value(gatewayResponse, "status");
        String gatewayPageUrl = value(
                gatewayResponse,
                "GatewayPageURL"
        );

        String sessionKey = value(
                gatewayResponse,
                "sessionkey"
        );

        if (!"SUCCESS".equalsIgnoreCase(gatewayStatus) ||
                !StringUtils.hasText(gatewayPageUrl) ||
                !StringUtils.hasText(sessionKey) ||
                !isTrustedGatewayUrl(gatewayPageUrl)) {

            payment.setStatus(PaymentStatus.FAILED);
            paymentRepository.save(payment);

            throw new ResponseStatusException(
                    HttpStatus.BAD_GATEWAY,
                    "SSLCommerz could not create a valid payment session."
            );
        }

        payment.setGatewaySessionKey(sessionKey);
        payment.setGatewayPageUrl(gatewayPageUrl);

        paymentRepository.save(payment);

        return toResponse(payment);
    }

public record CallbackResult(
        String orderNumber,
        PaymentStatus paymentStatus
) {}

@Transactional
public CallbackResult handleGatewayCallback(
        MultiValueMap<String, String> params,
        String callbackType
) {
    String transactionId = params.getFirst("tran_id");

    if (!StringUtils.hasText(transactionId)) {
        throw new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                "Missing transaction ID."
        );
    }

    Payment payment = paymentRepository
            .findByTransactionId(transactionId)
            .orElseThrow(() -> new ResponseStatusException(
                    HttpStatus.NOT_FOUND,
                    "Payment transaction not found."
            ));

    if (payment.getStatus() == PaymentStatus.PAID) {
        return callbackResult(payment);
    }

    String validationId = params.getFirst("val_id");

    if (StringUtils.hasText(validationId)) {
        return validateAndApply(payment, validationId);
    }

    return queryAndApply(payment);
}

private CallbackResult validateAndApply(
        Payment payment,
        String validationId
) {
    String endpoint = baseUrl.replaceAll("/+$", "")
            + "/validator/api/validationserverAPI.php";

    URI uri = UriComponentsBuilder
            .fromUriString(endpoint)
            .queryParam("val_id", validationId)
            .queryParam("store_id", storeId)
            .queryParam("store_passwd", storePassword)
            .queryParam("format", "json")
            .build()
            .encode()
            .toUri();

    Map<?, ?> result;

    try {
        result = restClient.get()
                .uri(uri)
                .accept(MediaType.APPLICATION_JSON)
                .retrieve()
                .body(Map.class);
    } catch (RestClientException exception) {
        throw new ResponseStatusException(
                HttpStatus.BAD_GATEWAY,
                "SSLCommerz validation service is unavailable."
        );
    }

    if (result == null ||
            !"DONE".equalsIgnoreCase(value(result, "APIConnect"))) {
        throw new ResponseStatusException(
                HttpStatus.BAD_GATEWAY,
                "Unable to verify the SSLCommerz transaction."
        );
    }

    String status = value(result, "status");

    if (!"VALID".equalsIgnoreCase(status) &&
            !"VALIDATED".equalsIgnoreCase(status)) {
        throw new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                "SSLCommerz transaction is not valid."
        );
    }

    if (!payment.getTransactionId().equals(
            value(result, "tran_id"))) {
        throw new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                "Transaction ID mismatch."
        );
    }

if (payment.getAmount().compareTo(
        new BigDecimal(value(result, "amount"))) != 0) {
        throw new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                "Payment amount mismatch."
        );
    }

    if (!"BDT".equalsIgnoreCase(value(result, "currency"))) {
        throw new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                "Payment currency mismatch."
        );
    }

    if (!payment.getOrder().getOrderNumber().equals(
            value(result, "value_a")) ||
            !payment.getTransactionId().equals(
                    value(result, "value_b"))) {
        throw new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                "Payment reference mismatch."
        );
    }

    // Risky payments must not be automatically confirmed.
    if (!"0".equals(value(result, "risk_level"))) {
        return callbackResult(payment);
    }

    payment.setGatewayValidationId(
            value(result, "val_id")
    );

    payment.setBankTransactionId(
            value(result, "bank_tran_id")
    );

    payment.setStatus(PaymentStatus.PAID);

    payment.getOrder().setStatus(OrderStatus.PENDING);

    paymentRepository.save(payment);
    orderRepository.save(payment.getOrder());

    return callbackResult(payment);
}

private CallbackResult queryAndApply(Payment payment) {

    String endpoint = baseUrl.replaceAll("/+$", "")
            + "/validator/api/merchantTransIDvalidationAPI.php";

    URI uri = UriComponentsBuilder
            .fromUriString(endpoint)
            .queryParam("tran_id", payment.getTransactionId())
            .queryParam("store_id", storeId)
            .queryParam("store_passwd", storePassword)
            .queryParam("format", "json")
            .build()
            .encode()
            .toUri();

    Map<?, ?> response;

    try {
        response = restClient.get()
                .uri(uri)
                .accept(MediaType.APPLICATION_JSON)
                .retrieve()
                .body(Map.class);
    } catch (RestClientException exception) {
        throw new ResponseStatusException(
                HttpStatus.BAD_GATEWAY,
                "Unable to query SSLCommerz transaction."
        );
    }

    if (response == null ||
            !"DONE".equalsIgnoreCase(
                    value(response, "APIConnect"))) {
        throw new ResponseStatusException(
                HttpStatus.BAD_GATEWAY,
                "SSLCommerz transaction query failed."
        );
    }

    Object elements = response.get("element");

    if (elements instanceof List<?> list) {
        for (Object element : list) {
            if (!(element instanceof Map<?, ?> transaction)) {
                continue;
            }

            if (!payment.getTransactionId().equals(
                    value(transaction, "tran_id"))) {
                continue;
            }

            String status = value(transaction, "status");

            if ("VALID".equalsIgnoreCase(status) ||
                    "VALIDATED".equalsIgnoreCase(status)) {

                String validationId = value(transaction, "val_id");

                if (StringUtils.hasText(validationId)) {
                    return validateAndApply(payment, validationId);
                }

            } else if ("FAILED".equalsIgnoreCase(status)) {

                payment.setStatus(PaymentStatus.FAILED);
                paymentRepository.save(payment);

            } else if ("CANCELLED".equalsIgnoreCase(status)) {

                payment.setStatus(PaymentStatus.CANCELLED);
                paymentRepository.save(payment);
            }

            break;
        }
    }

    return callbackResult(payment);
}

private CallbackResult callbackResult(Payment payment) {
    return new CallbackResult(
            payment.getOrder().getOrderNumber(),
            payment.getStatus()
    );
}


    private String callbackUrl(String action) {
        return callbackBaseUrl.replaceAll("/+$", "")
                + "/api/payments/callback/" + action;
    }

    private boolean isTrustedGatewayUrl(String gatewayPageUrl) {
        try {
            URI gatewayUri = URI.create(gatewayPageUrl);
            URI configuredUri = URI.create(baseUrl);

            return "https".equalsIgnoreCase(gatewayUri.getScheme())
                    && configuredUri.getHost() != null
                    && configuredUri.getHost().equalsIgnoreCase(
                            gatewayUri.getHost()
                    );

        } catch (IllegalArgumentException exception) {
            return false;
        }
    }

    private String value(Map<?, ?> response, String key) {
        Object result = response.get(key);
        return result == null ? "" : result.toString();
    }

    private PaymentInitiateResponse toResponse(Payment payment) {
        return new PaymentInitiateResponse(
                payment.getTransactionId(),
                payment.getAmount(),
                payment.getCurrency(),
                payment.getGatewayPageUrl(),
                payment.getStatus()
        );
    }
}

package com.khaidai.backend.service;

import com.khaidai.backend.dto.RecipeOverviewResponse;
import com.khaidai.backend.dto.RecipeRequests;

import com.khaidai.backend.entity.InventoryIngredient;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;

import java.math.BigDecimal;
import java.math.RoundingMode;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;

import java.util.List;

@Service
public class RecipeService {

    private final InventoryIngredientRepository ingredientRepository;
    private final MenuRecipeIngredientRepository recipeRepository;

    @Value("${khaidai.admin.api-key:}")
    private String configuredAdminKey;

    public RecipeService(
            InventoryIngredientRepository ingredientRepository,
            MenuRecipeIngredientRepository recipeRepository
    ) {
        this.ingredientRepository = ingredientRepository;
        this.recipeRepository = recipeRepository;
    }

    // ==========================================
    // GET RECIPE OVERVIEW
    // ==========================================

    @Transactional(readOnly = true)
    public RecipeOverviewResponse getOverview(String adminKey) {

        validateAdminKey(adminKey);

        List<RecipeOverviewResponse.MenuItemOption> menuItems =
                MenuCatalog.all()
                        .stream()
                        .map(food -> new RecipeOverviewResponse.MenuItemOption(
                                food.id(),
                                food.name(),
                                food.price()
                        ))
                        .toList();

        List<RecipeOverviewResponse.IngredientOption> ingredients =
                ingredientRepository.findAllByOrderByNameAsc()
                        .stream()
                        .filter(InventoryIngredient::isActive)
                        .map(ingredient ->
                                new RecipeOverviewResponse.IngredientOption(
                                        ingredient.getId(),
                                        ingredient.getName(),
                                        ingredient.getUnit(),
                                        ingredient.getCurrentStock()
                                )
                        )
                        .toList();

        List<RecipeOverviewResponse.RecipeRow> recipes =
                recipeRepository.findAllWithIngredient()
                        .stream()
                        .map(this::toRecipeRow)
                        .toList();

        return new RecipeOverviewResponse(
                menuItems,
                ingredients,
                recipes
        );
    }

    // ==========================================
    // ADD INGREDIENT TO RECIPE
    // ==========================================

    @Transactional
    public void addIngredient(
            Integer foodId,
            RecipeRequests.AddIngredientRequest request,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        if (request == null || request.ingredientId() == null) {
            throw badRequest("Please select an ingredient.");
        }

        MenuCatalog.FoodItem food = MenuCatalog.findById(foodId);

        if (food == null) {
            throw badRequest("Invalid menu item.");
        }

        InventoryIngredient ingredient = ingredientRepository
                .findById(request.ingredientId())
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Ingredient not found."
                ));

        if (!ingredient.isActive()) {
            throw badRequest("This ingredient is inactive.");
        }

        if (recipeRepository.findByFoodIdAndIngredient_Id(
                foodId,
                ingredient.getId()
        ).isPresent()) {

            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "This ingredient is already in the recipe."
            );
        }

        BigDecimal quantity = positiveQuantity(
                request.quantityPerServing()
        );

        MenuRecipeIngredient recipe = new MenuRecipeIngredient();

        recipe.setFoodId(food.id());
        recipe.setFoodName(food.name());
        recipe.setIngredient(ingredient);
        recipe.setQuantityPerServing(quantity);

        recipeRepository.save(recipe);
    }

    // ==========================================
    // UPDATE RECIPE QUANTITY
    // ==========================================

    @Transactional
    public void updateIngredient(
            Long recipeId,
            RecipeRequests.UpdateIngredientRequest request,
            String adminKey
    ) {

        validateAdminKey(adminKey);

        if (request == null) {
            throw badRequest("Recipe update information is required.");
        }

        MenuRecipeIngredient recipe = recipeRepository
                .findById(recipeId)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Recipe ingredient not found."
                ));

        recipe.setQuantityPerServing(
                positiveQuantity(request.quantityPerServing())
        );

        recipeRepository.save(recipe);
    }

    // ==========================================
    // REMOVE INGREDIENT FROM RECIPE
    // ==========================================

    @Transactional
    public void deleteIngredient(Long recipeId, String adminKey) {

        validateAdminKey(adminKey);

        MenuRecipeIngredient recipe = recipeRepository
                .findById(recipeId)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Recipe ingredient not found."
                ));

        recipeRepository.delete(recipe);
    }

    // ==========================================
    // RESPONSE MAPPING
    // ==========================================

    private RecipeOverviewResponse.RecipeRow toRecipeRow(
            MenuRecipeIngredient recipe
    ) {

        InventoryIngredient ingredient = recipe.getIngredient();

        return new RecipeOverviewResponse.RecipeRow(
                recipe.getId(),
                recipe.getFoodId(),
                recipe.getFoodName(),
                ingredient.getId(),
                ingredient.getName(),
                ingredient.getUnit(),
                ingredient.getCurrentStock(),
                recipe.getQuantityPerServing(),
                ingredient.isActive()
        );
    }

    // ==========================================
    // QUANTITY VALIDATION
    // ==========================================

    private BigDecimal positiveQuantity(BigDecimal value) {

        if (value == null || value.compareTo(BigDecimal.ZERO) <= 0) {
            throw badRequest("Quantity per serving must be greater than zero.");
        }

        BigDecimal normalized;

        try {
            normalized = value.setScale(3, RoundingMode.UNNECESSARY);
        } catch (ArithmeticException exception) {
            throw badRequest(
                    "Quantity supports a maximum of 3 decimal places."
            );
        }

        if (normalized.precision() > 14) {
            throw badRequest("Recipe quantity is too large.");
        }

        return normalized;
    }

    // ==========================================
    // ADMIN AUTHENTICATION
    // ==========================================

    private void validateAdminKey(String providedKey) {

        if (configuredAdminKey == null ||
                configuredAdminKey.isBlank()) {

            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "Admin API key is not configured."
            );
        }

        if (providedKey == null ||
                !MessageDigest.isEqual(
                        configuredAdminKey.getBytes(StandardCharsets.UTF_8),
                        providedKey.getBytes(StandardCharsets.UTF_8)
                )) {

            throw new ResponseStatusException(
                    HttpStatus.UNAUTHORIZED,
                    "Invalid administrator key."
            );
        }
    }

    private ResponseStatusException badRequest(String message) {

        return new ResponseStatusException(
                HttpStatus.BAD_REQUEST,
                message
        );
    }
}

import java.security.SecureRandom;
import java.util.Base64;

@Component
public class TableDataInitializer implements ApplicationRunner {

    private final TableRepository tableRepository;

    private static final SecureRandom SECURE_RANDOM =
            new SecureRandom();

    public TableDataInitializer(
            TableRepository tableRepository
    ) {
        this.tableRepository = tableRepository;
    }

    @Override
    public void run(ApplicationArguments args) {

        System.out.println(
                "Checking Khai-Dai restaurant tables..."
        );

        for (int tableNumber = 1; tableNumber <= 8; tableNumber++) {

            if (!tableRepository.existsByTableNumber(tableNumber)) {

                String token = generateSecureToken();

                RestaurantTable table = new RestaurantTable(
                        tableNumber,
                        token
                );

                tableRepository.save(table);

                System.out.println(
                        "Created restaurant table: " + tableNumber
                );
            }
        }

        System.out.println(
                "Khai-Dai table initialization completed."
        );
    }

    // Generate a cryptographically secure random token
    private String generateSecureToken() {

        byte[] randomBytes = new byte[32];

        SECURE_RANDOM.nextBytes(randomBytes);

        return Base64.getUrlEncoder()
                .withoutPadding()
                .encodeToString(randomBytes);
    }
}

    if (returnedOrderNumber) {
      return returnedOrderNumber;

      if (alreadySelected) {

        return current.filter((item) => item.id !== addon.id);

      return [...current, addon];

    food: Food,

    spiceLevel: SpiceLevel,

    addons: AddOn[]

    setOrderError("");

      a.id.localeCompare(b.id)

        (item) => getCartItemKey(item) === newKey

        return currentItems.map((item) =>

          getCartItemKey(item) === newKey

            ? { ...item, quantity: item.quantity + 1 }

            : item

      return [...currentItems, newItem];

  try {

    if (!orderNumber) {

        customerNote: customerNote.trim() || null,

        items: cartItems.map((item) => ({

        orderData = null;

orderNumber = confirmedOrderNumber;

      if (paymentMethod === "SSLCOMMERZ") {

async function getErrorMessage(response: Response): Promise<string> {
  const body = await response.text();

  if (!body) {
    return `Server error: ${response.status}`;
  }

  try {
    const data = JSON.parse(body);

    return (
      data.message ||
      data.detail ||
      data.error ||
      `Server error: ${response.status}`
    );
  } catch {
    return body;
  }
}

            name,
            unit,
            openingStock: Number(openingStock),
            reorderLevel: Number(reorderLevel),
          }),
        }
      );

      if (!response.ok) {
        throw new Error(await getErrorMessage(response));
      }

      setName("");
      setOpeningStock("0");
      setReorderLevel("1");

      await loadInventory();

    } catch (err) {

      setError(
        err instanceof Error
          ? err.message
          : "Unable to add ingredient."
      );

    } finally {

      setSaving(false);
    }
  };

  // ==========================================
  // RECORD STOCK MOVEMENT
  // ==========================================

  const handleMovement = async (
    event: FormEvent<HTMLFormElement>
  ) => {

    event.preventDefault();

    if (!movementAction) return;

    setSaving(true);
    setError("");

    try {

      const response = await fetch(
        `${API_BASE_URL}/api/admin/inventory/ingredients/${movementAction.id}/movements`,
        {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "X-Admin-Key": adminKey,
          },
          body: JSON.stringify({
            type: movementAction.type,
            quantity: Number(movementQuantity),
            note: movementNote,
          }),
        }
      );

      if (!response.ok) {
        throw new Error(await getErrorMessage(response));
      }

      setMovementAction(null);
      setMovementQuantity("");
      setMovementNote("");

      await loadInventory();

    } catch (err) {

      setError(
        err instanceof Error
          ? err.message
          : "Unable to record stock movement."
      );

    } finally {

      setSaving(false);
    }
  };

  // ==========================================
  // UPDATE REORDER LEVEL
  // ==========================================

  const saveReorderLevel = async (ingredient: Ingredient) => {

    const value = Number(
      reorderDrafts[ingredient.id] ??
      ingredient.reorderLevel
    );

    if (!Number.isFinite(value) || value < 0) {
      setError("Reorder level must be a non-negative number.");
      return;
    }

    setSaving(true);
    setError("");

    try {

      const response = await fetch(
        `${API_BASE_URL}/api/admin/inventory/ingredients/${ingredient.id}`,
        {
          method: "PATCH",
          headers: {
            "Content-Type": "application/json",
            "X-Admin-Key": adminKey,
          },
          body: JSON.stringify({
            reorderLevel: value,
          }),
        }
      );

      if (!response.ok) {
        throw new Error(await getErrorMessage(response));
      }

      await loadInventory();

    } catch (err) {

      setError(
        err instanceof Error
          ? err.message
          : "Unable to update reorder level."
      );

    } finally {

      setSaving(false);
    }
  };

  // ==========================================
  // ACTIVATE / DEACTIVATE
  // ==========================================

  const toggleActive = async (ingredient: Ingredient) => {

    const confirmed = window.confirm(
      ingredient.active
        ? `Deactivate ${ingredient.name}?`
        : `Reactivate ${ingredient.name}?`
    );

    if (!confirmed) return;

    setSaving(true);
    setError("");

    try {

      const response = await fetch(
        `${API_BASE_URL}/api/admin/inventory/ingredients/${ingredient.id}`,
        {
          method: "PATCH",
          headers: {
            "Content-Type": "application/json",
            "X-Admin-Key": adminKey,
          },
          body: JSON.stringify({
            active: !ingredient.active,
          }),
        }
      );

      if (!response.ok) {
        throw new Error(await getErrorMessage(response));
      }

      await loadInventory();

    } catch (err) {

      setError(
        err instanceof Error
          ? err.message
          : "Unable to update ingredient."
      );

    } finally {

      setSaving(false);
    }
  };

  const openMovement = (
    ingredient: Ingredient,
    type: MovementAction["type"]
  ) => {

    setMovementAction({
      id: ingredient.id,
      name: ingredient.name,
      unit: ingredient.unit,
      type,
    });

    setMovementQuantity("");
    setMovementNote("");
    setError("");
  };

  return (

    <section className="inventory-page">

      {/* HEADER */}

      <div className="inventory-heading">
        <div>
          <span className="manager-eyebrow">
            RESTAURANT STOCK CONTROL
          </span>

          <h2>Inventory Management</h2>

          <p>
            Manage ingredients, monitor stock and record stock movements.
          </p>
        </div>

        <button
          className="manager-refresh-button"
          onClick={() => void loadInventory()}
          disabled={loading}
        >
          {loading ? "Loading..." : "↻ Refresh"}
        </button>
      </div>

      {error && (
        <div className="manager-error">
          {error}
        </div>
      )}

      {/* SUMMARY */}

      <section className="inventory-summary">

        <article className="inventory-summary-card">
          <span>Active ingredients</span>
          <strong>{activeIngredients.length}</strong>
        </article>

        <article className="inventory-summary-card inventory-warning">
          <span>Low-stock ingredients</span>
          <strong>{lowStockCount}</strong>
        </article>

        <article className="inventory-summary-card inventory-danger">
          <span>Out of stock</span>
          <strong>{outOfStockCount}</strong>
        </article>

      </section>

      {/* ADD INGREDIENT */}

      <section className="inventory-panel">

        <div className="inventory-panel-heading">
          <div>
            <span className="manager-eyebrow">STOCK SETUP</span>
            <h3>Add new ingredient</h3>
          </div>
        </div>

        <form
          className="inventory-add-form"
          onSubmit={handleAddIngredient}
        >

          <div>
            <label htmlFor="ingredient-name">
              Ingredient name
            </label>

            <input
              id="ingredient-name"
              value={name}
              onChange={(event) => setName(event.target.value)}
              placeholder="e.g. Basmati rice"
              maxLength={100}
              required
            />
          </div>

          <div>
            <label htmlFor="ingredient-unit">Unit</label>

            <select
              id="ingredient-unit"
              value={unit}
              onChange={(event) => setUnit(event.target.value)}
            >
              <option value="kg">Kilogram (kg)</option>
              <option value="g">Gram (g)</option>
              <option value="L">Litre (L)</option>
              <option value="ml">Millilitre (ml)</option>
              <option value="pcs">Pieces (pcs)</option>
              <option value="pack">Pack</option>
              <option value="bottle">Bottle</option>
              <option value="box">Box</option>
            </select>
          </div>

          <div>
            <label htmlFor="opening-stock">Opening stock</label>

            <input
              id="opening-stock"
              type="number"
              min="0"
              step="0.001"
              value={openingStock}
              onChange={(event) => setOpeningStock(event.target.value)}
              required
            />
          </div>

          <div>
            <label htmlFor="reorder-level">Reorder level</label>

            <input
              id="reorder-level"
              type="number"
              min="0"
              step="0.001"
              value={reorderLevel}
              onChange={(event) => setReorderLevel(event.target.value)}
              required
            />
          </div>

          <button
            className="manager-primary-button"
            type="submit"
            disabled={saving}
          >
            {saving ? "Saving..." : "+ Add ingredient"}
          </button>

        </form>

      </section>

      {/* INGREDIENT TABLE */}

      <section className="inventory-panel">

        <div className="inventory-panel-heading">
          <div>
            <span className="manager-eyebrow">STOCK OVERVIEW</span>
            <h3>Ingredient inventory</h3>
          </div>

          <span className="inventory-count">
            {ingredients.length} ingredients
          </span>
        </div>

        {loading && !inventory ? (

          <div className="inventory-empty">
            Loading inventory...
          </div>

        ) : ingredients.length === 0 ? (

          <div className="inventory-empty">
            <span>📦</span>
            <h4>No ingredients added yet</h4>
            <p>Add your first ingredient using the form above.</p>
          </div>

        ) : (

          <div className="inventory-table-wrapper">

            <table className="inventory-table">

              <thead>
                <tr>
                  <th>Ingredient</th>
                  <th>Available stock</th>
                  <th>Reorder level</th>
                  <th>Status</th>
                  <th>Stock actions</th>
                </tr>
              </thead>

              <tbody>

                {ingredients.map((ingredient) => {

                  const outOfStock =
                    Number(ingredient.currentStock) <= 0;

                  return (

                    <tr key={ingredient.id}>

                      <td>
                        <strong>{ingredient.name}</strong>
                        <small>{ingredient.unit}</small>
                      </td>

                      <td>
                        <strong>
                          {formatQuantity(ingredient.currentStock)}
                        </strong>
                        {" "}{ingredient.unit}
                      </td>

                      <td>
                        <div className="inventory-reorder-control">

                          <input
                            type="number"
                            min="0"
                            step="0.001"
                            value={
                              reorderDrafts[ingredient.id] ??
                              String(ingredient.reorderLevel)
                            }
                            onChange={(event) =>
                              setReorderDrafts((previous) => ({
                                ...previous,
                                [ingredient.id]: event.target.value,
                              }))
                            }
                            disabled={!ingredient.active || saving}
                          />

                          <button
                            type="button"
                            onClick={() =>
                              void saveReorderLevel(ingredient)
                            }
                            disabled={!ingredient.active || saving}
                          >
                            Save
                          </button>

                        </div>
                      </td>

                      <td>

                        {!ingredient.active ? (

                          <span className="inventory-status inactive">
                            INACTIVE
                          </span>

                        ) : outOfStock ? (

                          <span className="inventory-status out">
                            OUT OF STOCK
                          </span>

                        ) : ingredient.lowStock ? (

                          <span className="inventory-status low">
                            LOW STOCK
                          </span>

                        ) : (

                          <span className="inventory-status good">
                            IN STOCK
                          </span>

                        )}

                      </td>

                      <td>

                        <div className="inventory-actions">

                          <button
                            type="button"
                            onClick={() => openMovement(ingredient, "STOCK_IN")}
                            disabled={!ingredient.active || saving}
                          >
                            Stock in
                          </button>

                          <button
                            type="button"
                            onClick={() => openMovement(ingredient, "WASTE")}
                            disabled={!ingredient.active || saving}
                          >
                            Waste
                          </button>

                          <button
                            type="button"
                            onClick={() => openMovement(ingredient, "ADJUSTMENT")}
                            disabled={!ingredient.active || saving}
                          >
                            Adjust
                          </button>

                          <button
                            type="button"
                            className="inventory-deactivate"
                            onClick={() => void toggleActive(ingredient)}
                            disabled={saving}
                          >
                            {ingredient.active ? "Deactivate" : "Reactivate"}
                          </button>

                        </div>

                      </td>

                    </tr>
                  );
                })}

              </tbody>

            </table>

          </div>

        )}

      </section>

      {/* MOVEMENT FORM */}

      {movementAction && (

        <section className="inventory-panel inventory-movement-form">

          <div className="inventory-panel-heading">

            <div>
              <span className="manager-eyebrow">
                STOCK TRANSACTION
              </span>

              <h3>
                {movementAction.type === "STOCK_IN"
                  ? "Record stock purchase"
                  : movementAction.type === "WASTE"
                    ? "Record wasted stock"
                    : "Adjust physical stock"}
              </h3>

              <p>
                {movementAction.name}
              </p>
            </div>

            <button
              type="button"
              className="manager-refresh-button"
              onClick={() => setMovementAction(null)}
            >
              Close
            </button>

          </div>

          <form onSubmit={handleMovement}>

            <div>
              <label htmlFor="movement-quantity">
                {movementAction.type === "ADJUSTMENT"
                  ? `Actual counted stock (${movementAction.unit})`
                  : `Quantity (${movementAction.unit})`}
              </label>

              <input
                id="movement-quantity"
                type="number"
                min="0"
                step="0.001"
                value={movementQuantity}
                onChange={(event) =>
                  setMovementQuantity(event.target.value)
                }
                required
              />

              {movementAction.type === "ADJUSTMENT" && (
                <small>
                  Enter the actual physical stock count. The system
                  calculates the difference automatically.
                </small>
              )}
            </div>

            <div>
              <label htmlFor="movement-note">
                Note / Reference
              </label>

              <input
                id="movement-note"
                value={movementNote}
                onChange={(event) => setMovementNote(event.target.value)}
                maxLength={250}
                placeholder="e.g. Supplier purchase or damaged stock"
              />
            </div>

            <button
              type="submit"
              className="manager-primary-button"
              disabled={saving}
            >
              {saving ? "Recording..." : "Confirm movement"}
            </button>

          </form>

        </section>

      )}

      {/* MOVEMENT HISTORY */}

      <section className="inventory-panel">

        <div className="inventory-panel-heading">
          <div>
            <span className="manager-eyebrow">AUDIT TRAIL</span>
            <h3>Recent stock movements</h3>
          </div>
        </div>

        {(inventory?.recentMovements ?? []).length === 0 ? (

          <div className="inventory-empty">
            No stock movements recorded yet.
          </div>

        ) : (

          <div className="inventory-table-wrapper">

            <table className="inventory-table">

              <thead>
                <tr>
                  <th>Date</th>
                  <th>Ingredient</th>
                  <th>Movement</th>
                  <th>Quantity change</th>
                  <th>Balance after</th>
                  <th>Note</th>
                </tr>
              </thead>

              <tbody>

                {inventory!.recentMovements.map((movement) => (

                  <tr key={movement.id}>

                    <td>{formatDate(movement.createdAt)}</td>

                    <td>{movement.ingredientName}</td>

                    <td>
                      <span className="inventory-movement-type">
                        {movement.movementType.replaceAll("_", " ")}
                      </span>
                    </td>

                    <td className={
                      Number(movement.quantityChange) >= 0
                        ? "movement-positive"
                        : "movement-negative"
                    }>
                      {Number(movement.quantityChange) > 0 ? "+" : ""}
                      {formatQuantity(movement.quantityChange)}
                      {" "}{movement.unit}
                    </td>

                    <td>
                      {formatQuantity(movement.balanceAfter)}
                      {" "}{movement.unit}
                    </td>

                    <td>{movement.note || "—"}</td>

                  </tr>

                ))}

              </tbody>

            </table>

          </div>

        )}

      </section>

    </section>
  );
}

const API_BASE_URL =
  `${window.location.protocol}//${window.location.hostname}:8090`;

const POLLING_INTERVAL = 12000;

async function getErrorMessage(response: Response): Promise<string> {
  const body = await response.text();

  if (!body) {
    return `Server error: ${response.status}`;
  }

  try {
    const data = JSON.parse(body);

    return (
      data.message ||
      data.error ||
      `Server error: ${response.status}`
    );
  } catch {
    return body;
  }
}

function formatMoney(amount: number): string {
  return `৳${Number(amount).toFixed(2)}`;
}

function formatDate(date: string): string {
  return new Date(date).toLocaleString("en-BD", {
    dateStyle: "medium",
    timeStyle: "short",
  });
}

function getNextAction(status: OrderStatus) {
  switch (status) {
    case "PENDING":
      return {
        label: "Accept order",
        nextStatus: "ACCEPTED" as OrderStatus,
      };

    case "ACCEPTED":
      return {
        label: "Start preparing",
        nextStatus: "PREPARING" as OrderStatus,
      };

    case "PREPARING":
      return {
        label: "Mark as ready",
        nextStatus: "READY" as OrderStatus,
      };

    case "READY":
      return {
        label: "Mark as served",
        nextStatus: "SERVED" as OrderStatus,
      };

    default:
      return null;
  }
}

        `${API_BASE_URL}/api/admin/recipes/foods/${selectedFoodId}/ingredients`,
        {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "X-Admin-Key": adminKey,
          },
          body: JSON.stringify({
            ingredientId: Number(ingredientId),
            quantityPerServing: amount,
          }),
        }
      );

      if (!response.ok) {
        throw new Error(await getErrorMessage(response));
      }

      setIngredientId("");
      setQuantity("");

      await loadRecipes();

    } catch (err) {

      setError(
        err instanceof Error
          ? err.message
          : "Unable to add recipe ingredient."
      );

    } finally {

      setSaving(false);
    }
  };

  // ==========================================
  // UPDATE RECIPE QUANTITY
  // ==========================================

  const handleUpdateQuantity = async (recipe: RecipeRow) => {

    const amount = Number(
      quantityDrafts[recipe.id] ??
      recipe.quantityPerServing
    );

    if (!Number.isFinite(amount) || amount <= 0) {
      setError("Quantity must be greater than zero.");
      return;
    }

    setSaving(true);
    setError("");

    try {

      const response = await fetch(
        `${API_BASE_URL}/api/admin/recipes/${recipe.id}`,
        {
          method: "PATCH",
          headers: {
            "Content-Type": "application/json",
            "X-Admin-Key": adminKey,
          },
          body: JSON.stringify({
            quantityPerServing: amount,
          }),
        }
      );

      if (!response.ok) {
        throw new Error(await getErrorMessage(response));
      }

      await loadRecipes();

    } catch (err) {

      setError(
        err instanceof Error
          ? err.message
          : "Unable to update recipe quantity."
      );

    } finally {

      setSaving(false);
    }
  };

  // ==========================================
  // REMOVE INGREDIENT
  // ==========================================

  const handleRemoveIngredient = async (recipe: RecipeRow) => {

    const confirmed = window.confirm(
      `Remove ${recipe.ingredientName} from ${recipe.foodName}?`
    );

    if (!confirmed) return;

    setSaving(true);
    setError("");

    try {

      const response = await fetch(
        `${API_BASE_URL}/api/admin/recipes/${recipe.id}`,
        {
          method: "DELETE",
          headers: {
            "X-Admin-Key": adminKey,
          },
        }
      );

      if (!response.ok) {
        throw new Error(await getErrorMessage(response));
      }

      await loadRecipes();

    } catch (err) {

      setError(
        err instanceof Error
          ? err.message
          : "Unable to remove recipe ingredient."
      );

    } finally {

      setSaving(false);
    }
  };

  // ==========================================
  // USER INTERFACE
  // ==========================================

  return (

function localDateString(date: Date): string {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, "0");
  const day = String(date.getDate()).padStart(2, "0");

  return `${year}-${month}-${day}`;
}

function getInitialRange(): DateRange {
  const today = new Date();
  const firstDay = new Date();

  firstDay.setDate(today.getDate() - 6);

  return {
    from: localDateString(firstDay),
    to: localDateString(today),
  };
}

function numberValue(value: number | string | null | undefined): number {
  return Number(value ?? 0);
}

function formatMoney(value: number | string): string {
  return `৳${numberValue(value).toFixed(2)}`;
}

function formatChartDate(date: string): string {
  return new Date(`${date}T00:00:00`).toLocaleDateString("en-BD", {
    day: "numeric",
    month: "short",
  });
}

async function getErrorMessage(response: Response): Promise<string> {
  const body = await response.text();

  if (!body) {
    return `Server error: ${response.status}`;
  }

  try {
    const data = JSON.parse(body);

    return (
      data.message ||
      data.detail ||
      data.error ||
      `Server error: ${response.status}`
    );
  } catch {
    return body;
  }
}

import { useEffect, useState } from "react";

const API_BASE_URL =
  `${window.location.protocol}//${window.location.hostname}:8090`;

type Session = {
  status: "checking" | "valid" | "invalid";
  tableNumber?: number;
  message?: string;
};

type ValidationResponse = {
  valid: boolean;
  tableNumber: number;
  message: string;
};

export default function TableSessionBanner() {
  const [session, setSession] = useState<Session | null>(null);

  useEffect(() => {
    const token = new URLSearchParams(
      window.location.search
    ).get("tableToken");

    if (!token) {
      setSession(null);
      return;
    }

    const controller = new AbortController();

    const validateTable = async () => {
      setSession({ status: "checking" });

      try {
        const response = await fetch(
          `${API_BASE_URL}/api/tables/validate/${encodeURIComponent(token)}`,
          { signal: controller.signal }
        );

        if (!response.ok) {
          throw new Error(
            "Invalid or inactive restaurant table QR code."
          );
        }

        const data: ValidationResponse =
          await response.json();

        if (!data.valid) {
          throw new Error("Table verification failed.");
        }

        setSession({
          status: "valid",
          tableNumber: data.tableNumber,
          message: data.message,
        });

      } catch (err) {
        if (
          err instanceof Error &&
          err.name === "AbortError"
        ) {
          return;
        }

        setSession({
          status: "invalid",
          message:
            err instanceof Error
              ? err.message
              : "Unable to verify table.",
        });
      }
    };

    validateTable();

    return () => controller.abort();
  }, []);

  if (!session) return null;

  if (session.status === "checking") {
    return (
      <div className="table-session-banner checking">
        Verifying your restaurant table...
      </div>
    );
  }

  if (session.status === "invalid") {
    return (
      <div className="table-session-banner invalid">
        ⚠️ {session.message}
      </div>
    );
  }

  return (
    <div className="table-session-banner valid">
      <span>✓</span>

      <div>
        <strong>
          Welcome! You are ordering from Table{" "}
          {session.tableNumber}.
        </strong>

        <small>
          Your table has been verified successfully.
        </small>
      </div>
    </div>
  );
}
